  ingredients:
    - { amount: "1/2 lb", item: "ground beef" }
    - { amount: "1/2 lb", item: "ground pork" }
    - { amount: "1/4 cup", item: "breadcrumbs" }
    - { amount: "1/4 cup", item: "grated Parmesan" }
    - { amount: "1", item: "egg" }
    - { amount: "1 tsp", item: "garlic powder" }
    - { amount: "1 tbsp", item: "olive oil" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "2", item: "carrots, diced" }
    - { amount: "2", item: "celery stalks, diced" }
    - { amount: "8 cups", item: "chicken broth" }
    - { amount: "3/4 cup", item: "acini di pepe or small pasta" }
    - { amount: "4 cups", item: "baby spinach or escarole" }
  steps:
    - "Mix the beef, pork, breadcrumbs, Parmesan, egg and garlic powder and roll into small meatballs about 3/4-inch wide."
    - "Cook the onion, carrots and celery in the oil for 5 minutes. Add the broth and bring to a simmer."
    - "Add the meatballs and simmer 10 minutes. Stir in the pasta and cook 8 minutes until tender."
    - "Add the greens and cook until wilted. Serve with extra Parmesan."
