  ingredients:
    - { amount: "2 tbsp", item: "extra-virgin olive oil" }
    - { amount: "1", item: "medium yellow onion, chopped" }
    - { amount: "1 tsp", item: "cumin seeds" }
    - { amount: "1 tsp", item: "garam masala" }
    - { amount: "1/2 tsp", item: "ground coriander" }
    - { amount: "1/2 tsp", item: "ground turmeric" }
    - { amount: "1/4 tsp", item: "ground cardamom" }
    - { amount: "Pinch", item: "cayenne pepper" }
    - { amount: "2", item: "garlic cloves, grated" }
    - { amount: "1/2 tsp", item: "fresh ginger, grated" }
    - { amount: "Heaping 1 tsp", item: "sea salt" }
    - { amount: "1/2 cup", item: "water, divided" }
    - { amount: "1 can (28 oz)", item: "whole peeled tomatoes" }
    - { amount: "3 cups", item: "cooked chickpeas (two 15-oz cans), drained and rinsed" }
    - { amount: "1/2 cup", item: "fresh cilantro, chopped" }
    - { amount: "1 tsp", item: "lemon juice" }
    - { amount: "To serve", item: "rice, Greek yogurt, extra cilantro and lemon" }
  steps:
    - "Heat the oil in a large skillet over medium heat and cook the onion about 8 minutes, stirring often, until soft and well browned."
    - "Lower the heat and add the cumin, garam masala, coriander, turmeric, cardamom and cayenne. Stir for 30 seconds until fragrant, then stir in the garlic, ginger, salt and 1/4 cup of the water."
    - "Add the tomatoes, crushing them as you go, and simmer 8-10 minutes until the sauce thickens."
    - "Add the chickpeas and the remaining water and simmer 20-25 minutes, stirring occasionally, until thick."
    - "Stir in the cilantro and lemon juice. Serve over rice with yogurt, more cilantro and lemon."
note: "Adapted from Love and Lemons' chana masala (4.92 stars from 49 votes)."
