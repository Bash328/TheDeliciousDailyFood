description: "Golden broth, tender chicken, vegetables and egg noodles."

recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 30
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "2 tbsp", item: "olive oil" }
    - { amount: "1", item: "large onion, diced" }
    - { amount: "3", item: "carrots, sliced" }
    - { amount: "3", item: "celery stalks, sliced" }
    - { amount: "4", item: "garlic cloves, minced" }
    - { amount: "8 cups", item: "chicken broth" }
    - { amount: "1 1/2 lbs", item: "boneless chicken breasts or thighs" }
    - { amount: "2", item: "bay leaves" }
    - { amount: "1 tsp", item: "dried thyme" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
    - { amount: "2 cups", item: "egg noodles" }
    - { amount: "2 tbsp", item: "fresh parsley, chopped" }
  steps:
    - "Heat the oil in a large pot and cook the onion, carrots and celery for 6 minutes. Add the garlic for 30 seconds."
    - "Pour in the broth, add the chicken, bay leaves, thyme, salt and pepper, and bring to a boil. Reduce the heat and simmer 15-20 minutes until the chicken is cooked through."
    - "Remove the chicken, shred it with two forks, and return it to the pot."
    - "Add the noodles and cook until tender, 6-8 minutes."
    - "Remove the bay leaves, stir in the parsley and serve."
note: "For leftovers, cook the noodles separately so they don't swell and soak up the broth."
