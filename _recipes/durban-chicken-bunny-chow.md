recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 45
  servingsNumber: 4
  servingsUnit: Bunnies
  ingredients:
    - { amount: "2 lbs", item: "boneless, skinless chicken thighs, diced" }
    - { amount: "2 tbsp", item: "oil" }
    - { amount: "2", item: "medium onions, diced" }
    - { amount: "2", item: "cinnamon sticks" }
    - { amount: "2", item: "bay leaves" }
    - { amount: "5", item: "curry leaves (optional)" }
    - { amount: "1 tsp", item: "cumin seeds" }
    - { amount: "5", item: "cardamom pods, crushed" }
    - { amount: "4 tsp", item: "Durban masala or curry powder" }
    - { amount: "2 tsp", item: "garam masala" }
    - { amount: "1 tsp", item: "turmeric" }
    - { amount: "1 tsp", item: "ground coriander" }
    - { amount: "2 tsp", item: "garlic, chopped" }
    - { amount: "2 tsp", item: "fresh ginger, chopped" }
    - { amount: "1 can (14 oz)", item: "chopped tomatoes" }
    - { amount: "1 tbsp", item: "sugar" }
    - { amount: "1 1/2 to 2 tsp", item: "salt" }
    - { amount: "1 1/2 to 2 cups", item: "water" }
    - { amount: "1-2", item: "green chilies, chopped (or a pinch of cayenne)" }
    - { amount: "2", item: "unsliced square sandwich loaves" }
    - { amount: "To serve", item: "fresh coriander and carrot sambal" }
  steps:
    - "Heat the oil in a large pot over medium-high heat. Add the onions, cinnamon, bay leaves, curry leaves, cumin seeds and cardamom and cook about 5 minutes until the onions are soft and just turning golden."
    - "Add the Durban masala, garam masala, turmeric, coriander, garlic and ginger and cook 1 minute, scraping the bottom so nothing burns."
    - "Stir in the tomatoes and cook about 3 minutes until darker and thicker. Add the sugar and salt."
    - "Add the chicken and coat it in the sauce for 2-3 minutes. Pour in enough water to almost cover the chicken and bring to a gentle simmer."
    - "Cover and simmer on low about 30 minutes until the chicken is tender, stirring now and then. If it's too saucy at the halfway point, take the lid off so it can thicken."
    - "Cut each loaf in half, or quarters for smaller portions. Hollow out each piece, leaving a border of about 2 cm on the sides and bottom, and save the soft middle for dipping."
    - "Spoon the hot curry into the bread, top with coriander and serve right away with carrot sambal and the bread middles."
note: "Adapted from The Cinnamon Jar's bunny chow (rated 5 stars from 3 votes, a small sample). Durban masala gives the most authentic flavour; use medium curry powder if you can't find it."
