recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 10
  servingsNumber: 4
  servingsUnit: Sandwiches
  ingredients:
    - { amount: "4", item: "Cuban or hoagie rolls, split" }
    - { amount: "3 tbsp", item: "yellow mustard" }
    - { amount: "1 lb", item: "roast pork, thinly sliced" }
    - { amount: "1/2 lb", item: "deli ham" }
    - { amount: "8 slices", item: "Swiss cheese" }
    - { amount: "To taste", item: "dill pickle slices" }
    - { amount: "2 tbsp", item: "butter, softened" }
  steps:
    - "Spread mustard inside each roll."
    - "Layer the pork, ham, Swiss and pickles in each roll."
    - "Butter the outside of the rolls and press in a hot skillet, weighted with a heavy pan or a sandwich press, 3-4 minutes per side until golden and the cheese has melted."
    - "Cut diagonally and serve hot."
