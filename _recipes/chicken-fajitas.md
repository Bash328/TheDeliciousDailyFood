  servingsUnit: Servings
  ingredients:
    - { amount: "2 tbsp", item: "fresh lime juice" }
    - { amount: "2 tbsp", item: "oil" }
    - { amount: "1", item: "large garlic clove, minced" }
    - { amount: "1 1/2 tsp", item: "ground cumin" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2-1 tsp", item: "ground chili" }
    - { amount: "1 tbsp", item: "fresh cilantro, chopped (optional)" }
    - { amount: "1 1/2 lbs", item: "boneless, skinless chicken thighs" }
    - { amount: "3", item: "large bell peppers (mixed colors), cut into strips" }
    - { amount: "1", item: "red onion, thinly sliced" }
    - { amount: "2", item: "avocados, peeled and sliced" }
    - { amount: "To serve", item: "warm tortillas, cilantro and your favorite fillings" }
  steps:
    - "Combine the lime juice, oil, garlic, cumin, salt, chili and cilantro in a shallow bowl. Add the chicken and marinate 30 minutes if you have time."
    - "Heat a large 12-inch skillet over medium-high heat until smoking and add a drizzle of oil."
    - "Sear the chicken on both sides, flipping a couple of times, about 8 minutes per side, until golden, charred and cooked through."
    - "Remove the chicken, add the peppers and onion to the pan and cook until the onion is soft and the peppers lightly charred. Season with salt and pepper."
    - "Slice the chicken into strips."
    - "Serve with warm tortillas, cilantro and avocado."
