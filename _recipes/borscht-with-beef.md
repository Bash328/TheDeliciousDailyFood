image: ""
recipeContent:
  prepTimeMinutes: 30
  cookTimeMinutes: 90
  servingsNumber: 8
  servingsUnit: Bowls
  ingredients:
    - { amount: "1 lb", item: "beef sirloin or stew meat, cut into 1-inch pieces" }
    - { amount: "14 cups", item: "cold water" }
    - { amount: "1 tbsp", item: "salt, plus more to taste" }
    - { amount: "2 large", item: "beets, peeled and grated (or 3 medium)" }
    - { amount: "4 tbsp", item: "olive oil" }
    - { amount: "1 tbsp", item: "vinegar" }
    - { amount: "1 tbsp", item: "sugar" }
    - { amount: "2 tbsp", item: "tomato sauce or paste" }
    - { amount: "1 tbsp", item: "unsalted butter" }
    - { amount: "1", item: "medium onion, finely diced" }
    - { amount: "2", item: "carrots, grated" }
    - { amount: "2", item: "large potatoes, peeled and sliced into bite-sized pieces" }
    - { amount: "1/2", item: "small head cabbage, sliced" }
    - { amount: "2", item: "tomatoes, peeled and diced" }
    - { amount: "2", item: "bay leaves" }
    - { amount: "1/4 tsp", item: "freshly ground pepper" }
    - { amount: "1/4 cup", item: "chopped fresh parsley, plus more for garnish" }
    - { amount: "2", item: "garlic cloves, pressed" }
    - { amount: "To serve", item: "sour cream and fresh parsley or dill" }
  steps:
    - "Wash the beef and put it in a large pot with the cold water and 1 tbsp salt. Bring to a boil and skim off the foam as soon as it forms. Reduce the heat, partially cover and simmer 45 minutes to 1 hour, skimming occasionally."
    - "Sauté the grated beets in a heavy skillet with the olive oil and vinegar for 5 minutes. Lower the heat, add the sugar and tomato sauce and cook about 10 minutes until starting to soften. Set aside."
    - "In the same skillet, cook the onion in the butter 2 minutes, add the carrots and cook about 5 minutes until soft."
    - "Once the meat has simmered at least 45 minutes, add the potatoes and cook 10 minutes."
    - "Add the cabbage, beets, onion-carrot mixture and tomatoes and cook another 10 minutes, until the potatoes are tender."
    - "Add the bay leaves, pepper and more salt to taste."
    - "Stir in the parsley and garlic, cover and take off the heat. Rest 20 minutes. Serve with sour cream and herbs."
note: "Adapted from Natasha's Kitchen's borscht with meat (4.96 stars from 253 votes). It's even better the next day."
