recipeContent:
  prepTimeMinutes: 40
  cookTimeMinutes: 25
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "10 oz (300 g)", item: "English spinach leaves" }
    - { amount: "1/2 tsp", item: "salt, for the spinach" }
    - { amount: "6 oz (175 g)", item: "Greek feta, crumbled" }
    - { amount: "1/2 cup", item: "Greek yogurt" }
    - { amount: "2", item: "green onions, finely sliced" }
    - { amount: "1", item: "egg" }
    - { amount: "2 tbsp", item: "finely chopped mint" }
    - { amount: "1 tbsp", item: "finely chopped dill" }
    - { amount: "1/2 tsp", item: "lemon zest" }
    - { amount: "1/2 tbsp", item: "lemon juice" }
    - { amount: "2", item: "garlic cloves, finely chopped" }
    - { amount: "1/2 tsp", item: "freshly grated nutmeg" }
    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "cayenne pepper" }
    - { amount: "16 sheets", item: "filo pastry" }
    - { amount: "8 tbsp (120 g)", item: "butter, melted" }
    - { amount: "2/3 cup", item: "finely grated kefalotyri or Parmesan" }
    - { amount: "1 tsp", item: "white and black sesame seeds, mixed" }
  steps:
    - "Heat the oven to 430°F and line a tray with baking paper."
    - "Toss the spinach with 1/2 tsp salt and leave 10 minutes. Wring it out in handfuls in a tea towel to get rid of the water."
    - "Mix the spinach with the feta, yogurt, green onions, egg, herbs, lemon zest and juice, garlic, nutmeg, pepper, salt and cayenne."
    - "Trim the filo to about 13 x 10 inches. Layer 8 sheets on the tray, brushing each with melted butter."
    - "Spread the filling over the filo, leaving a 1-inch border."
    - "Cover with a filo sheet, brush with butter and sprinkle with the cheese. Repeat for 5 sheets in all, then add 3 more sheets with butter between them but no cheese."
    - "Press the edges to seal (crimp if you like), brush the top with butter and sprinkle with sesame seeds."
    - "Bake 25 minutes until golden. Serve straight away, while the base is still crisp."
note: "Adapted from RecipeTin Eats' spanakopita (4.93 stars from 85 votes)."
