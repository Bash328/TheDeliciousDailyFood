image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 45
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "1 tbsp", item: "coconut oil" }
    - { amount: "1/2", item: "onion, thinly sliced" }
    - { amount: "2", item: "garlic cloves, chopped" }
    - { amount: "1/2", item: "red jalapeño, sliced (or 1-3 Thai chiles, halved)" }
    - { amount: "3", item: "slices galangal (1/4-inch), or ginger" }
    - { amount: "1", item: "lemongrass stalk, pounded and cut into 2-inch pieces" }
    - { amount: "2 tsp", item: "red Thai curry paste" }
    - { amount: "4 cups", item: "chicken broth" }
    - { amount: "4 cups", item: "unsweetened full-fat coconut cream or coconut milk" }
    - { amount: "2", item: "medium chicken breasts, cut into bite-sized pieces (about 1 lb)" }
    - { amount: "8 oz", item: "white mushrooms, sliced" }
    - { amount: "1-2 tbsp", item: "coconut sugar" }
    - { amount: "1 1/2-2 tbsp", item: "fish sauce, plus more to taste" }
    - { amount: "2-3 tbsp", item: "fresh lime juice" }
    - { amount: "2-3", item: "green onions, thinly sliced" }
    - { amount: "To garnish", item: "fresh cilantro" }
  steps:
    - "Heat the coconut oil in a medium pot over medium heat. Add the onion, garlic, chile, galangal, lemongrass and curry paste and cook, stirring often, 5 minutes until the onion softens."
    - "Add the broth, bring to a boil, reduce the heat and simmer uncovered 30 minutes."
    - "Strain out the aromatics and discard them. Add the coconut cream, chicken and mushrooms and simmer until the chicken is just cooked through."
    - "Add the fish sauce, coconut sugar and lime juice and taste, adding more of each as needed."
    - "Cook 2 minutes, then ladle into bowls and top with green onions and cilantro."
note: "Adapted from 40 Aprons' tom kha gai (4.91 stars from 798 votes). Tofu or shrimp work in place of the chicken."
