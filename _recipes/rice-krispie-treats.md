recipeContent:
  prepTimeMinutes: 5

  servingsUnit: Bars
  ingredients:
    - { amount: "4 tbsp", item: "butter" }
    - { amount: "10 oz", item: "marshmallows (about 40 regular)" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "6 cups", item: "crispy rice cereal" }
    - { amount: "1 tsp", item: "vanilla extract" }
  steps:
    - "Grease a 9x13-inch pan."
    - "Melt the butter in a large pot over low heat. Add the marshmallows and stir until completely melted."
    - "Remove from the heat and stir in the vanilla and salt, then the cereal until evenly coated."
    - "Press into the pan with a greased spatula. Don't pack too tightly, or the treats get hard."
    - "Cool 30 minutes and cut into squares."
