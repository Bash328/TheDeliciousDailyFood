recipeContent:
  prepTimeMinutes: 30
  cookTimeMinutes: 10
  servingsNumber: 6
  servingsUnit: Servings
  ingredients:
    - { amount: "3/4 cup", item: "extra-virgin olive oil, divided" }
    - { amount: "6 cups", item: "rustic Italian bread, cut into 1-inch cubes (about 3/4 lb)" }
    - { amount: "1 tsp", item: "salt, divided" }
    - { amount: "1 1/2 lbs", item: "tomatoes, cut into 1-inch cubes" }
    - { amount: "2", item: "red, yellow or orange bell peppers, cut into 1-inch cubes" }
    - { amount: "1", item: "small English cucumber, seeded and thinly sliced" }
    - { amount: "3 tbsp", item: "capers, drained" }
    - { amount: "3 tbsp", item: "white wine vinegar" }
    - { amount: "1", item: "large garlic clove, minced" }
    - { amount: "3 tbsp", item: "minced shallots" }
    - { amount: "1/4 tsp", item: "freshly ground black pepper" }
    - { amount: "1/3 cup", item: "fresh basil, chopped" }
  steps:
    - "Heat 1/4 cup of the oil in a large skillet over medium heat. Add the bread and 1/2 tsp salt and cook, stirring often, about 10 minutes until crisp and golden. Cool."
    - "In a large bowl, toss the tomatoes, peppers, cucumber, capers, vinegar, garlic, shallots, pepper, remaining 1/2 tsp salt and remaining 1/2 cup oil."
    - "Right before serving, add two-thirds of the bread and the basil and toss. Taste and adjust the salt."
    - "Transfer to a serving bowl and scatter the rest of the bread on top. Serve immediately so the bread stays crisp."
note: "Adapted from Once Upon a Chef's panzanella (4.98 stars from 36 votes). Ripe, in-season tomatoes make it."
