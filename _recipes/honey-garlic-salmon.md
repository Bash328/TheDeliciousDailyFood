prepTimeMinutes: 5
cookTimeMinutes: 15
servingsNumber: 2
servingsUnit: Servings
ingredients:
  - { amount: "2", item: "salmon fillets, skin on" }
  - { amount: "1/2 tsp", item: "salt" }
  - { amount: "1 tbsp", item: "olive oil" }
  - { amount: "3 tbsp", item: "honey" }
  - { amount: "2 tbsp", item: "soy sauce" }
  - { amount: "3", item: "garlic cloves, minced" }
  - { amount: "1 tbsp", item: "rice vinegar" }
  - { amount: "1", item: "scallion, sliced, for serving" }
steps:
  - "Pat the salmon dry and season with salt. Heat olive oil in a pan over medium-high heat."
  - "Sear the salmon skin-side down for 4 minutes, flip, and cook 3 more minutes until just cooked through. Remove from the pan."
  - "Whisk honey, soy sauce, garlic, and rice vinegar together. Pour into the same pan and simmer 2-3 minutes until it thickens into a glaze."
  - "Return the salmon to the pan, spooning the glaze over the top for a minute to coat."
  - "Top with sliced scallion and serve over rice."
note: "Swap the salmon for chicken thighs using the same glaze and timing — it works just as well."
