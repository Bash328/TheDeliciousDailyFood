recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 15
  servingsNumber: 2
  servingsUnit: Servings
  ingredients:
    - { amount: "2 tbsp", item: "butter, divided" }
    - { amount: "8 oz", item: "cremini mushrooms, sliced" }
    - { amount: "1", item: "small shallot, finely chopped" }
    - { amount: "2", item: "garlic cloves, minced" }
    - { amount: "1/4 cup", item: "cream cheese, softened" }
    - { amount: "2 tbsp", item: "sour cream" }
    - { amount: "1", item: "roasted red pepper, sliced into strips" }
    - { amount: "", item: "Salt and black pepper, to taste" }
    - { amount: "4", item: "thick slices sourdough bread" }
    - { amount: "1 cup", item: "baby arugula" }
    - { amount: "1 tbsp", item: "olive oil, for toasting" }
  steps:
    - "Melt 1 tbsp butter in a skillet over medium-high heat. Add the mushrooms and cook, stirring occasionally, 6-8 minutes until deeply golden. Add the shallot and garlic and cook 1-2 minutes more until softened."
    - "Remove from heat and stir in the cream cheese and sour cream until the mushrooms are coated in a thick, creamy filling. Season with salt and pepper."
    - "Divide the mushroom mixture and roasted red pepper strips between two slices of bread, and top with a handful of arugula. Close the sandwiches with the remaining bread."
    - "Wipe out the skillet, melt the remaining butter with the olive oil, and toast the sandwiches 3-4 minutes per side, pressing gently, until deeply golden and crisp."
    - "Slice in half and serve warm."
note: "The mushroom filling keeps well in the fridge for two days — reheat gently before assembling."
