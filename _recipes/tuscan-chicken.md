prepTimeMinutes: 10
cookTimeMinutes: 25
servingsNumber: 4
servingsUnit: Servings
ingredients:
  - { amount: "4", item: "boneless chicken breasts, halved lengthwise" }
  - { amount: "1 tsp", item: "salt" }
  - { amount: "1 tsp", item: "Italian seasoning" }
  - { amount: "2 tbsp", item: "olive oil" }
  - { amount: "3", item: "garlic cloves, minced" }
  - { amount: "1 cup", item: "heavy cream" }
  - { amount: "1/2 cup", item: "chicken broth" }
  - { amount: "1/3 cup", item: "sun-dried tomatoes, chopped" }
  - { amount: "1/2 cup", item: "grated parmesan" }
  - { amount: "2 cups", item: "baby spinach" }
steps:
  - "Season the chicken with salt and Italian seasoning. Sear in olive oil over medium-high heat, 5-6 minutes per side, until golden. Remove and set aside."
  - "Lower the heat, add garlic to the same pan, and cook 30 seconds until fragrant."
  - "Pour in the cream and broth, scraping up the browned bits. Stir in sun-dried tomatoes and parmesan, simmer 3-4 minutes until it thickens slightly."
  - "Stir in the spinach until just wilted, about a minute."
  - "Return the chicken to the pan, spoon sauce over the top, and simmer 5 minutes until the chicken is cooked through."
note: "Serve over pasta or with crusty bread — the sauce is the whole point."
