  prepTimeMinutes: 15

  servingsNumber: 3
  servingsUnit: Servings
  ingredients:
    - { amount: "1 1/2 cups (2 medium)", item: "potatoes, cut into 3/4-inch cubes" }
    - { amount: "2 cups (180 g)", item: "cauliflower florets, about 1 1/2 inches" }
    - { amount: "3/4 to 1 cup", item: "finely chopped onion" }
    - { amount: "3/4 to 1 cup", item: "finely chopped tomatoes" }
    - { amount: "1/2 tbsp", item: "minced ginger" }
    - { amount: "1/2 tbsp", item: "minced garlic" }
    - { amount: "1", item: "green chili, slit or chopped (optional)" }
    - { amount: "2 tbsp", item: "chopped coriander leaves" }
    - { amount: "2 to 3 tbsp", item: "oil" }
    - { amount: "1/2 tsp", item: "cumin seeds" }
    - { amount: "3/4 to 1 1/4 tsp", item: "Kashmiri red chili powder" }
    - { amount: "1/4 tsp", item: "turmeric" }
    - { amount: "1 tsp", item: "garam masala" }
    - { amount: "3/4 to 1 tsp", item: "coriander powder" }
    - { amount: "1/2 to 3/4 tsp", item: "roasted cumin powder" }
    - { amount: "1 tbsp", item: "kasuri methi (dried fenugreek leaves)" }
    - { amount: "1/2 to 1 tsp", item: "amchur (dried mango powder, optional)" }
    - { amount: "1/2 to 3/4 tsp", item: "salt" }
    - { amount: "To serve", item: "lemon juice, rice or roti" }
  steps:
    - "Soak the cauliflower in slightly hot water for 3-4 minutes, then drain, rinse and drain well. Keep the cubed potatoes in water until you need them."
    - "Heat the oil and add the cumin seeds. When they sizzle, cook the ginger and garlic 30 seconds, then the onion until transparent."
    - "Add the drained potatoes and green chili and stir-fry 2-3 minutes. Cover and cook on low to medium heat until half done, adding a splash of water if needed."
    - "Add the cauliflower and stir-fry about 3 minutes."
    - "Add the chili powder, turmeric, cumin powder, garam masala and coriander powder, and sprinkle over 2-3 tbsp of water to keep it moist."
    - "Cook covered, stirring every few minutes. Add the salt and cook until the potatoes are soft and the cauliflower is just tender but still slightly crunchy."
    - "Add the tomatoes and kasuri methi and fry on medium-high until the raw tomato smell is gone, 2-4 minutes."
    - "Finish with amchur, coriander leaves and a squeeze of lemon. Serve with rice or roti."
