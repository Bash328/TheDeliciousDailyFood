    - { amount: "2", item: "chicken breasts" }

    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "4", item: "large flour tortillas" }
    - { amount: "3 cups", item: "romaine lettuce, chopped" }
    - { amount: "1/2 cup", item: "grated Parmesan" }
    - { amount: "1/2 cup", item: "Caesar dressing" }
    - { amount: "1/2 cup", item: "croutons, crushed" }
  steps:
    - "Season the chicken with salt and pepper and cook in the oil over medium-high heat 5-6 minutes per side to 165°F. Rest 5 minutes and slice."
    - "Toss the romaine with the dressing, Parmesan and croutons."
    - "Lay the tortillas flat, divide the salad and chicken among them, fold the sides in and roll up tightly."
    - "Slice in half to serve."
