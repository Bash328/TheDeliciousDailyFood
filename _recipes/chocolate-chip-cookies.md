date: 2026-01-01
category: Dessert
cuisine: "American"
description: "Chewy centers, crisp edges and plenty of melty chocolate."
image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 12
  servingsNumber: 24
  servingsUnit: Cookies
  ingredients:
    - { amount: "2 1/4 cups", item: "all-purpose flour" }
    - { amount: "1 tsp", item: "baking soda" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1 cup", item: "butter, softened" }
    - { amount: "3/4 cup", item: "granulated sugar" }
    - { amount: "3/4 cup", item: "packed brown sugar" }
    - { amount: "2", item: "large eggs" }
    - { amount: "2 tsp", item: "vanilla extract" }
    - { amount: "2 cups", item: "semi-sweet chocolate chips" }
  steps:
    - "Heat the oven to 375°F. Whisk the flour, baking soda and salt together."
    - "Beat the butter with both sugars until fluffy, about 3 minutes. Beat in the eggs one at a time, then the vanilla."
    - "Stir in the flour mixture until just combined, then fold in the chocolate chips."
    - "Scoop rounded tablespoons onto ungreased baking sheets, 2 inches apart."
    - "Bake 9-11 minutes until the edges are golden and the centers still look slightly soft."
    - "Cool on the pan for 5 minutes, then move to a rack."
note: "For thicker, bakery-style cookies, chill the dough for an hour before baking."
