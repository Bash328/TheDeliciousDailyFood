recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 90
  servingsNumber: 8
  servingsUnit: Bowls
  ingredients:
    - { amount: "2 tbsp", item: "olive oil" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "2", item: "carrots, diced" }
    - { amount: "2", item: "celery stalks, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 lb", item: "dried green split peas, rinsed" }
    - { amount: "1", item: "meaty ham bone or 2 cups diced ham" }
    - { amount: "8 cups", item: "chicken broth or water" }
    - { amount: "2", item: "bay leaves" }
    - { amount: "1/2 tsp", item: "dried thyme" }
    - { amount: "1/2 tsp", item: "black pepper" }
  steps:
    - "Cook the onion, carrots and celery in the oil for 6 minutes. Add the garlic for 30 seconds."
    - "Add the split peas, ham bone, broth, bay leaves, thyme and pepper and bring to a boil."
    - "Reduce the heat and simmer, partially covered, 1 1/2 to 2 hours, stirring occasionally, until the peas have broken down."
    - "Remove the bone and bay leaves, shred the meat from the bone back into the soup, and season to taste."
note: "Split peas don't need soaking."
