title: "Bacon and Egg Breakfast Casserole"
slug: bacon-and-egg-breakfast-casserole
date: 2026-01-01
category: Breakfast
cuisine: "American"
description: "A make-ahead bake with eggs, bacon, cheese and hash browns."
image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 50
  servingsNumber: 8
  servingsUnit: Servings
  ingredients:
    - { amount: "1 lb", item: "bacon, cooked and crumbled" }
    - { amount: "20 oz", item: "frozen shredded hash browns, thawed" }
    - { amount: "2 cups", item: "shredded cheddar" }
    - { amount: "10", item: "large eggs" }
    - { amount: "1 1/2 cups", item: "milk" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
    - { amount: "1/2 tsp", item: "mustard powder" }
    - { amount: "4", item: "scallions, sliced" }
  steps:
    - "Heat the oven to 375°F and grease a 9x13-inch baking dish."
    - "Spread the hash browns in the dish and top with the bacon, 1 1/2 cups of the cheese and the scallions."
    - "Whisk the eggs, milk, salt, pepper and mustard powder and pour over everything."
    - "Bake 40-45 minutes, sprinkling with the remaining cheese for the last 5 minutes, until set in the center."
    - "Rest 10 minutes before cutting."
note: "Assemble the night before and refrigerate, then bake in the morning."
---
