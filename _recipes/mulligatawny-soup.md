recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 30
  servingsNumber: 4
  servingsUnit: Bowls
  ingredients:
    - { amount: "2 tbsp", item: "vegetable oil" }
    - { amount: "1 tbsp", item: "garam masala (or curry powder)" }
    - { amount: "1 cup", item: "diced carrots (2-3 carrots)" }
    - { amount: "1 cup", item: "chopped yellow onion" }
    - { amount: "1/2 cup", item: "finely chopped celery" }
    - { amount: "1", item: "tart-sweet apple (such as Honeycrisp or Fuji), peeled and finely chopped" }
    - { amount: "6 cups", item: "chicken broth" }
    - { amount: "1 cup", item: "canned unsweetened coconut milk" }
    - { amount: "1 1/3 cups", item: "red lentils" }
    - { amount: "Scant 1/2 tsp", item: "salt" }
    - { amount: "2 cups", item: "shredded cooked chicken" }
    - { amount: "1 tbsp", item: "fresh lime juice, plus lime wedges" }
    - { amount: "Pinch", item: "cayenne pepper (optional)" }
    - { amount: "1/2 cup", item: "fresh cilantro, chopped, for serving" }
  steps:
    - "Heat the oil in a large soup pot over medium heat. Stir in the garam masala for 30 seconds, then add the carrots, onion, celery and apple and cook 7-8 minutes, stirring often, until soft."
    - "Add the broth, coconut milk, lentils and salt. Bring to a boil, reduce to low and simmer uncovered about 15 minutes until the lentils are tender and breaking down."
    - "Puree with an immersion blender until slightly thickened (or blend 2-3 cups and stir back in)."
    - "Add the chicken and lime juice and heat through. Season with more salt, lime and cayenne."
    - "Serve topped with cilantro and lime wedges."
note: "Adapted from Once Upon a Chef's mulligatawny (4.84 stars from 100 votes). A rotisserie chicken makes it quick."
