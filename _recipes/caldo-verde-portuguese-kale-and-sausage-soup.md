    - { amount: "2 tbsp", item: "olive oil" }
    - { amount: "1", item: "yellow onion, chopped" }
    - { amount: "4", item: "garlic cloves, minced" }
    - { amount: "1 1/2 lbs", item: "Yukon Gold potatoes, peeled, quartered and thinly sliced" }
    - { amount: "4 cups", item: "finely chopped kale (or collard or mustard greens)" }
    - { amount: "8 cups", item: "chicken broth" }
    - { amount: "1/4 cup", item: "dry sherry" }
    - { amount: "1", item: "lemon, zested" }
    - { amount: "3/4 tsp", item: "smoked paprika" }
    - { amount: "1/2 tsp", item: "salt, plus more to taste" }
    - { amount: "12 oz", item: "hard Portuguese chouriço, linguiça or Spanish chorizo" }
    - { amount: "15 oz", item: "white beans, drained (1 can)" }
    - { amount: "1/4 cup", item: "heavy cream" }
    - { amount: "To taste", item: "pepper" }
  steps:
    - "Heat the oil in a 6-quart heavy pot over medium heat and cook the onion and garlic 3 minutes."
    - "Add the potatoes, kale, broth, sherry, lemon zest, smoked paprika and 1/2 tsp salt. Bring to a boil, then lower the heat, cover and simmer 15 minutes until the potatoes and kale are soft."
    - "Peel the casing off the sausage, quarter the links lengthwise and slice into thin wedges."
    - "Stir in the sausage, white beans and cream and simmer 5 more minutes. Taste and season with salt and pepper."
