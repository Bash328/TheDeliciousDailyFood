recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 10
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "1 1/2 lbs", item: "ground beef (80/20)" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
    - { amount: "4 slices", item: "American or cheddar cheese" }
    - { amount: "4", item: "hamburger buns" }
    - { amount: "1 tbsp", item: "butter, softened" }
    - { amount: "4", item: "lettuce leaves" }
    - { amount: "1", item: "tomato, sliced" }
    - { amount: "1/2", item: "red onion, sliced" }
    - { amount: "To serve", item: "dill pickle slices" }
  steps:
    - "Divide the beef into 4 loose balls, flatten to patties slightly wider than the buns, and press a shallow dimple in the middle of each."
    - "Season both sides with salt and pepper just before cooking."
    - "Cook in a hot cast-iron skillet or grill 4 minutes per side for medium. Add the cheese in the last minute and cover so it melts."
    - "Butter the buns and toast them cut-side down in the pan until golden."
    - "Build with lettuce, tomato, onion, pickles and your sauce of choice."
note: "Handle the meat as little as possible. Pressed-together patties turn out dense."
