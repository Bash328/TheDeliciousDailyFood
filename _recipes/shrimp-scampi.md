recipeContent:
  prepTimeMinutes: 2
  cookTimeMinutes: 8
  servingsNumber: 6
  servingsUnit: Servings
  ingredients:
    - { amount: "1 1/2 lbs", item: "large shrimp (16-20 count), peeled and deveined" }
    - { amount: "1/2 tsp", item: "salt, plus more to taste" }
    - { amount: "1/2 tsp", item: "black pepper, plus more to taste" }
    - { amount: "2 tbsp", item: "unsalted butter, divided" }
    - { amount: "2 tbsp", item: "extra virgin olive oil, divided" }
    - { amount: "4", item: "garlic cloves, minced" }
    - { amount: "1/4 tsp", item: "red pepper flakes (optional)" }
    - { amount: "1/2 cup", item: "dry white wine (or low-sodium chicken broth)" }
    - { amount: "1 1/2 tbsp", item: "lemon juice, plus wedges for serving" }
    - { amount: "1/4 cup", item: "chopped fresh parsley" }
    - { amount: "To serve", item: "pasta or crusty bread" }
  steps:
    - "Pat the shrimp dry and season with 1/2 tsp salt and 1/2 tsp pepper."
    - "Heat 1 tbsp oil and 1 tbsp butter in a large skillet over medium-high heat. Cook the shrimp in a single layer 1-2 minutes per side until just opaque. Remove right away so they don't overcook."
    - "Lower the heat to medium and add the remaining oil and butter. Cook the garlic and pepper flakes about 30 seconds."
    - "Add the wine and simmer 2 minutes until reduced by half, then stir in the lemon juice."
    - "Return the shrimp and their juices to the pan and toss to coat. Take off the heat and stir in the parsley. Adjust the salt and pepper."
    - "Serve over pasta or with crusty bread and lemon wedges."
note: "Adapted from Natasha's Kitchen's shrimp scampi (4.98 stars from 118 votes)."
