    - { amount: "1/4 tsp", item: "fine sea salt" }
    - { amount: "1 cup", item: "butter lettuce, torn, ribs removed" }
    - { amount: "1 cup", item: "red cabbage, very thinly sliced" }
    - { amount: "2", item: "medium carrots, cut into matchsticks" }
    - { amount: "2", item: "mini cucumbers, thinly sliced or cut into strips" }
    - { amount: "2", item: "medium jalapeños, seeded and thinly sliced" }
    - { amount: "1/4 cup", item: "green onions, thinly sliced" }
    - { amount: "1/4 cup", item: "fresh cilantro, chopped" }
    - { amount: "1/4 cup", item: "fresh mint, chopped" }
    - { amount: "8 sheets", item: "rice paper wrappers" }
    - { amount: "1/3 cup", item: "creamy peanut butter" }
    - { amount: "2 tbsp", item: "rice vinegar" }
    - { amount: "2 tbsp", item: "reduced-sodium tamari or soy sauce" }
    - { amount: "2 tbsp", item: "honey or maple syrup" }
    - { amount: "1 tbsp", item: "toasted sesame oil, for the sauce" }
    - { amount: "2", item: "garlic cloves, minced" }
    - { amount: "2-3 tbsp", item: "water, as needed" }
  steps:
    - "Cook the noodles just until al dente, drain, rinse with cool water, then toss with 1 tsp sesame oil and the salt."
    - "Whisk the peanut butter, vinegar, tamari, honey, 1 tbsp sesame oil and garlic, adding water until creamy but dippable."
    - "Fill a shallow pan with an inch of water and set a folded towel beside it. Mix the green onions, cilantro and mint."
    - "Dip a rice paper in the water for about 20 seconds until pliable, then lay it flat on the towel."
    - "Leaving a 1-inch border, layer lettuce, noodles, cabbage, carrot, cucumber and jalapeño on the lower third and sprinkle with the herbs."
    - "Fold the bottom up over the filling, tuck in the sides like a burrito and roll up tightly. Repeat with the remaining wrappers."
    - "Serve whole or sliced in half on the diagonal, with the peanut sauce."
