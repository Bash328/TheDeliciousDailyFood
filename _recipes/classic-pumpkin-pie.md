recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 60
  servingsNumber: 8
  servingsUnit: Slices
  ingredients:
    - { amount: "1", item: "refrigerated pie crust (9-inch)" }
    - { amount: "1 can (15 oz)", item: "pumpkin purée" }
    - { amount: "1 can (12 oz)", item: "evaporated milk" }
    - { amount: "2", item: "large eggs" }
    - { amount: "3/4 cup", item: "granulated sugar" }
    - { amount: "1 tsp", item: "ground cinnamon" }
    - { amount: "1/2 tsp", item: "ground ginger" }
    - { amount: "1/4 tsp", item: "ground cloves" }
    - { amount: "1/2 tsp", item: "salt" }
  steps:
    - "Heat the oven to 425°F and fit the crust into a pie plate."
    - "Whisk the pumpkin, evaporated milk, eggs, sugar, cinnamon, ginger, cloves and salt until smooth."
    - "Pour into the crust and bake 15 minutes."
    - "Reduce the heat to 350°F and bake 40-50 minutes more until a knife near the center comes out clean."
    - "Cool completely before slicing, then serve with whipped cream."
