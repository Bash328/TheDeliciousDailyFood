date: 2026-01-01
category: Breakfast
cuisine: "American"
description: "Bakery-style muffins with a tall golden top and bursts of blueberry."
image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 25
  servingsNumber: 12
  servingsUnit: Muffins
  ingredients:
    - { amount: "2 cups", item: "all-purpose flour" }
    - { amount: "3/4 cup", item: "granulated sugar" }
    - { amount: "2 tsp", item: "baking powder" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "1/2 cup", item: "vegetable oil" }
    - { amount: "2", item: "large eggs" }
    - { amount: "3/4 cup", item: "milk" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "1 1/2 cups", item: "blueberries (fresh or frozen)" }
    - { amount: "2 tbsp", item: "coarse sugar, for topping" }
  steps:
    - "Heat the oven to 425°F and line a 12-cup muffin tin."
    - "Whisk the flour, sugar, baking powder and salt. In another bowl whisk the oil, eggs, milk and vanilla."
    - "Stir the wet mixture into the dry until just combined, then gently fold in the blueberries."
    - "Fill the cups almost to the top and sprinkle with coarse sugar."
    - "Bake 5 minutes at 425°F, then reduce to 350°F and bake 15-18 minutes more until golden."
note: "The high starting temperature gives the muffins their tall domed tops."
