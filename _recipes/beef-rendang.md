image: ""
recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 180
  servingsNumber: 6
  servingsUnit: Servings
  ingredients:
    - { amount: "12", item: "dried chilies, rehydrated in boiling water (or 12 large fresh chilies)" }
    - { amount: "1", item: "small onion, finely chopped" }
    - { amount: "5", item: "garlic cloves, minced" }
    - { amount: "3", item: "lemongrass stalks, white part only, sliced" }
    - { amount: "1 1/2 tbsp", item: "fresh galangal, finely chopped" }
    - { amount: "1 1/2 tbsp", item: "fresh ginger, minced" }
    - { amount: "2 tbsp", item: "vegetable oil" }
    - { amount: "2 lbs", item: "chuck steak, cut into 1.6-inch cubes" }
    - { amount: "1 tbsp", item: "vegetable oil, for browning" }
    - { amount: "1", item: "cinnamon stick" }
    - { amount: "1/4 tsp", item: "clove powder" }
    - { amount: "3", item: "star anise" }
    - { amount: "1/2 tsp", item: "cardamom powder" }
    - { amount: "1", item: "lemongrass stick, bottom half, smashed" }
    - { amount: "14 oz", item: "coconut milk (1 can)" }
    - { amount: "2 tsp", item: "tamarind paste" }
    - { amount: "4", item: "large kaffir lime leaves, very finely sliced" }
    - { amount: "1/3 cup", item: "desiccated coconut" }
    - { amount: "1 tbsp", item: "brown sugar or grated palm sugar" }
    - { amount: "1 1/2 tsp", item: "salt" }
  steps:
    - "Blitz the chilies, onion, garlic, lemongrass, galangal, ginger and 2 tbsp oil in a food processor into a fine paste."
    - "Heat 1 tbsp oil in a large heavy pot over high heat and brown the beef in two batches. Set aside."
    - "Lower the heat to medium-low and cook the spice paste 2-3 minutes until it darkens and dries out a little."
    - "Add the cinnamon, clove, star anise, cardamom, smashed lemongrass, coconut milk, tamarind, lime leaves, desiccated coconut, sugar, salt and the beef."
    - "Bring to a simmer, then turn the heat low so it bubbles very gently. Cover and simmer 1 hour 15 minutes."
    - "Uncover and check the beef: it should be quite tender but not falling apart yet."
    - "Raise the heat to medium and cook 30-40 minutes, stirring more often toward the end, until the sauce reduces to a thick paste coating the beef and the meat is very tender. Add a splash of water if it's not tender yet."
    - "Serve with rice or coconut rice."
note: "Adapted from RecipeTin Eats' beef rendang (4.97 stars from 231 votes). The long reduction at the end is what makes it a true rendang."
