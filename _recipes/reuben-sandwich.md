recipeContent:
  prepTimeMinutes: 10

  servingsNumber: 2
  servingsUnit: Sandwiches
  ingredients:
    - { amount: "4 slices", item: "rye bread" }
    - { amount: "2 tbsp", item: "butter, softened" }
    - { amount: "1/4 cup", item: "Russian or Thousand Island dressing" }
    - { amount: "1/2 lb", item: "corned beef, thinly sliced" }
    - { amount: "4 slices", item: "Swiss cheese" }
    - { amount: "1/2 cup", item: "sauerkraut, drained and squeezed dry" }
  steps:
    - "Butter one side of each slice of bread. Spread dressing on the other sides."
    - "In a skillet over medium heat, place two slices butter-side down. Layer cheese, corned beef, sauerkraut and more cheese."
    - "Top with the remaining bread, butter-side up."
    - "Cook 4-5 minutes per side, pressing down, until golden and the cheese has melted."
