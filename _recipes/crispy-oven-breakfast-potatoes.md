  ingredients:
    - { amount: "2 lbs", item: "Yukon Gold potatoes, peeled and cut into 1/2-inch cubes" }
    - { amount: "3 tbsp", item: "extra-light olive oil (or another high-heat oil)" }
    - { amount: "1 tsp", item: "sea salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "1 tsp", item: "paprika" }
    - { amount: "1 tbsp", item: "parsley, finely chopped (optional)" }
  steps:
    - "Heat the oven to 400°F and line a large rimmed baking sheet with parchment or a silicone mat."
    - "Toss the potatoes with the oil, paprika, salt and pepper until evenly coated."
    - "Spread in a single layer and bake 20 minutes until cooked through."
    - "Stir quickly, switch the oven to broil and cook 2-4 minutes more until golden brown. Sprinkle with parsley."
