recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 15
  servingsNumber: 4
  servingsUnit: Pancakes
  ingredients:
    - { amount: "1 1/2 cups", item: "all-purpose flour" }
    - { amount: "2 tbsp", item: "brown sugar" }
    - { amount: "2 tsp", item: "baking powder" }
    - { amount: "1/2 tsp", item: "cinnamon" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "1 cup", item: "milk" }
    - { amount: "1", item: "large egg" }
    - { amount: "2 tbsp", item: "melted butter" }
    - { amount: "2", item: "ripe bananas, mashed" }
    - { amount: "1 tsp", item: "vanilla extract" }
  steps:
    - "Whisk the flour, sugar, baking powder, cinnamon and salt."
    - "Mix the milk, egg, butter, bananas and vanilla, then stir into the dry ingredients until just combined."
    - "Cook 1/4 cup batter per pancake on a greased skillet over medium heat 2-3 minutes per side until golden."
    - "Serve with sliced banana and syrup."
