cuisine: "American"
description: "A creamy chicken and vegetable filling under a flaky golden crust."
image: ""
recipeContent:
  prepTimeMinutes: 30
  cookTimeMinutes: 40
  servingsNumber: 6
  servingsUnit: Servings
  ingredients:
    - { amount: "2", item: "refrigerated pie crusts (9-inch)" }
    - { amount: "4 tbsp", item: "butter" }
    - { amount: "1", item: "small onion, diced" }
    - { amount: "2", item: "carrots, diced" }
    - { amount: "2", item: "celery stalks, diced" }
    - { amount: "1/3 cup", item: "all-purpose flour" }
    - { amount: "1 3/4 cups", item: "chicken broth" }
    - { amount: "3/4 cup", item: "whole milk" }
    - { amount: "3 cups", item: "cooked chicken, shredded" }
    - { amount: "1 cup", item: "frozen peas" }
    - { amount: "1 tsp", item: "dried thyme" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
    - { amount: "1", item: "egg, beaten, for brushing" }
  steps:
    - "Heat the oven to 400°F. Fit one crust into a 9-inch pie dish."
    - "Melt the butter in a pot over medium heat. Cook the onion, carrots and celery for 6 minutes until soft."
    - "Stir in the flour and cook 1 minute. Slowly whisk in the broth and milk and simmer 3-4 minutes until thick."
    - "Stir in the chicken, peas, thyme, salt and pepper. Pour the filling into the crust."
    - "Cover with the second crust, trim and crimp the edges, and cut a few slits in the top. Brush with the beaten egg."
    - "Bake 35-40 minutes until deep golden. Rest 15 minutes before slicing."
note: "A rotisserie chicken makes this a weeknight dinner."
