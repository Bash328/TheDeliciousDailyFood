slug: vanilla-pound-cake
date: 2026-01-01
category: Dessert
cuisine: "American"
description: "A buttery, dense and tender cake that keeps well for days."
image: ""
recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 70
  servingsNumber: 12
  servingsUnit: Slices
  ingredients:
    - { amount: "1 cup", item: "butter, softened" }
    - { amount: "1 3/4 cups", item: "granulated sugar" }
    - { amount: "4", item: "large eggs" }
    - { amount: "2 tsp", item: "vanilla extract" }
    - { amount: "2 cups", item: "all-purpose flour" }
    - { amount: "1/2 tsp", item: "baking powder" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "1/2 cup", item: "milk" }
  steps:
    - "Heat the oven to 325°F and grease and flour a 9x5-inch loaf pan."
    - "Beat the butter and sugar until pale and fluffy, 4 minutes. Beat in the eggs one at a time, then the vanilla."
    - "Whisk the flour, baking powder and salt. Add to the batter alternately with the milk, mixing until just combined."
    - "Pour into the pan and bake 65-75 minutes until a toothpick comes out clean."
    - "Cool in the pan for 15 minutes, then turn out onto a rack."
