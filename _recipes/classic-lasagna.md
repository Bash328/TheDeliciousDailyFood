cuisine: "Italian"
description: "Layers of meat sauce, ricotta and mozzarella baked until bubbling."
image: ""
recipeContent:
  prepTimeMinutes: 40
  cookTimeMinutes: 60
  servingsNumber: 10
  servingsUnit: Servings
  ingredients:
    - { amount: "1 lb", item: "ground beef" }
    - { amount: "1/2 lb", item: "Italian sausage, casings removed" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 jar (24 oz)", item: "marinara sauce" }
    - { amount: "1 can (15 oz)", item: "tomato sauce" }
    - { amount: "15 oz", item: "ricotta cheese" }
    - { amount: "1", item: "egg" }
    - { amount: "1/2 cup", item: "grated Parmesan" }
    - { amount: "2 tbsp", item: "fresh parsley, chopped" }
    - { amount: "12", item: "lasagna noodles, cooked" }
    - { amount: "4 cups", item: "shredded mozzarella" }
  steps:
    - "Heat the oven to 375°F. Brown the beef and sausage with the onion in a large pot, 10 minutes. Add the garlic, marinara and tomato sauce and simmer 15 minutes."
    - "Mix the ricotta, egg, half the Parmesan and the parsley."
    - "Spread a thin layer of sauce in a 9x13-inch dish. Layer 4 noodles, one third of the ricotta, one third of the sauce and a quarter of the mozzarella."
    - "Repeat twice. Finish with the last noodles, remaining sauce, mozzarella and Parmesan."
    - "Cover with foil and bake 30 minutes, then uncover and bake 25 minutes more until golden."
    - "Rest 15 minutes before cutting so the layers hold."
note: "Assemble it the day before, refrigerate, then add 10 minutes to the covered baking time."
