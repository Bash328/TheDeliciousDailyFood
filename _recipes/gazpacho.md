recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 0
  servingsNumber: 4
  servingsUnit: Bowls
  ingredients:
    - { amount: "1", item: "English cucumber" }
    - { amount: "2 1/2 lbs", item: "ripe tomatoes, chopped" }
    - { amount: "2", item: "Fresno chiles (or 1/2 red bell pepper), stemmed and seeded" }
    - { amount: "1/4", item: "small red onion, rinsed" }
    - { amount: "2", item: "garlic cloves" }
    - { amount: "1/4 cup", item: "chopped fresh cilantro, plus more for garnish" }
    - { amount: "3 tbsp", item: "sherry vinegar or red wine vinegar" }
    - { amount: "1/2 cup", item: "extra virgin olive oil, plus more for drizzling" }
    - { amount: "1 1/4 tsp", item: "sea salt" }
    - { amount: "1/4 tsp", item: "freshly ground black pepper" }
    - { amount: "To garnish", item: "cherry tomatoes and fresh herbs" }
  steps:
    - "Finely chop a quarter of the cucumber and set it aside for garnish."
    - "Peel the rest, cut into chunks, and blend with the tomatoes, chiles, onion, garlic, cilantro, vinegar, olive oil, salt and pepper until smooth."
    - "Season to taste and chill at least 2 hours."
    - "Serve cold, garnished with the diced cucumber, cherry tomatoes, herbs, a drizzle of olive oil and black pepper."
note: "Adapted from Love and Lemons' gazpacho (4.95 stars from 200 votes). Use the ripest tomatoes you can find."
