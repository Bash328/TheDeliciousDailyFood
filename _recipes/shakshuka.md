  cookTimeMinutes: 25
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "2 tbsp", item: "extra-virgin olive oil" }
    - { amount: "1", item: "small white onion, diced" }
    - { amount: "1", item: "red bell pepper, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 tsp", item: "ground cumin" }
    - { amount: "1/2 tsp", item: "paprika" }
    - { amount: "Pinch", item: "cayenne pepper (optional)" }
    - { amount: "1 can (28 oz)", item: "fire-roasted crushed tomatoes" }
    - { amount: "1/2 tsp", item: "sea salt, plus more to taste" }
    - { amount: "To taste", item: "freshly ground black pepper" }
    - { amount: "6", item: "large eggs" }
    - { amount: "1/4 cup", item: "fresh parsley or cilantro leaves" }
    - { amount: "1/4 cup", item: "crumbled feta cheese" }
    - { amount: "To serve", item: "pita" }
  steps:
    - "Heat the oil in a large lidded skillet over medium heat. Cook the onion and pepper 5-8 minutes until soft. Add the garlic, cumin, paprika and cayenne and stir 30 seconds."
    - "Add the tomatoes, salt and plenty of pepper and simmer on low, stirring often, about 15 minutes until thickened."
    - "Make 6 wells in the sauce and crack an egg into each. Cover and cook 4-8 minutes, depending on how runny you like them."
    - "Season to taste, top with parsley and feta and serve with pita."
