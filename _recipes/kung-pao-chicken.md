  ingredients:
    - { amount: "12 oz", item: "boneless, skinless chicken breasts, cut into small cubes" }
    - { amount: "1 tbsp", item: "cornstarch" }
    - { amount: "1 tbsp", item: "Shaoxing rice wine (optional)" }
    - { amount: "2 tsp", item: "soy sauce" }
    - { amount: "1 tsp", item: "oil" }
    - { amount: "1 1/2 tbsp", item: "soy sauce, for the sauce" }
    - { amount: "1 tsp", item: "dark soy sauce" }
    - { amount: "1 tsp", item: "sugar" }
    - { amount: "1/4 tsp", item: "Chinese black vinegar" }
    - { amount: "2 tbsp", item: "water" }
    - { amount: "1 tsp", item: "cornstarch, for the sauce" }
    - { amount: "3 tbsp", item: "oil, divided" }
    - { amount: "2", item: "garlic cloves, sliced" }
    - { amount: "5 slices", item: "fresh ginger" }
    - { amount: "6-8", item: "dried red chilies, seeded and halved" }
    - { amount: "3 tbsp", item: "roasted peanuts" }
    - { amount: "1 stalk", item: "scallion, cut into rings" }
  steps:
    - "Pat the chicken dry and marinate with the cornstarch, rice wine, 2 tsp soy sauce and 1 tsp oil for 30 minutes."
    - "Stir the sauce ingredients together in a small bowl."
    - "Heat 1 tbsp oil in a wok and stir-fry the chicken until the surface is white and about 70% cooked. Remove it."
    - "Wipe the wok, heat the remaining 2 tbsp oil and quickly stir-fry the ginger and garlic. Add the dried chilies and stir until fragrant."
    - "Return the chicken, stir briefly, then add the peanuts."
    - "Pour in the sauce and stir until the chicken is coated. Add the scallions, toss and serve immediately with steamed rice."
