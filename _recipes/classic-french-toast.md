    - { amount: "4", item: "large eggs" }
    - { amount: "1 cup", item: "whole milk" }
    - { amount: "2 tsp", item: "vanilla extract" }
    - { amount: "1 tsp", item: "ground cinnamon" }
    - { amount: "1 tbsp", item: "granulated sugar" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "8 slices", item: "thick bread (brioche or challah is best)" }
    - { amount: "2 tbsp", item: "butter, for cooking" }
    - { amount: "To serve", item: "maple syrup and berries" }
  steps:
    - "Whisk the eggs, milk, vanilla, cinnamon, sugar and salt in a shallow dish."
    - "Dip each slice of bread in the mixture for about 10 seconds per side, letting it soak without falling apart."
    - "Melt a little butter in a skillet over medium heat. Cook the bread 2-3 minutes per side until deep golden."
    - "Serve hot with syrup and berries."
note: "Slightly stale bread soaks up the custard better than fresh."
