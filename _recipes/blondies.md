cuisine: "American"
description: "Chewy brown sugar bars with butterscotch flavor and chocolate chips."
image: ""
recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 28
  servingsNumber: 16
  servingsUnit: Bars
  ingredients:
    - { amount: "1/2 cup", item: "butter, melted" }
    - { amount: "1 cup", item: "packed brown sugar" }
    - { amount: "1", item: "large egg" }
    - { amount: "2 tsp", item: "vanilla extract" }
    - { amount: "1 cup", item: "all-purpose flour" }
    - { amount: "1/2 tsp", item: "baking powder" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "3/4 cup", item: "chocolate chips or white chocolate chips" }
  steps:
    - "Heat the oven to 350°F and line an 8-inch square pan with parchment."
    - "Whisk the melted butter and brown sugar until smooth. Beat in the egg and vanilla."
    - "Stir in the flour, baking powder and salt until just combined, then fold in the chips."
    - "Spread in the pan and bake 25-28 minutes until golden and the center is set."
    - "Cool before cutting."
