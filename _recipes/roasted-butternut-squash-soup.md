    - { amount: "1", item: "large butternut squash (about 3 lbs), peeled and cubed" }
    - { amount: "3 tbsp", item: "olive oil" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "4 cups", item: "vegetable or chicken broth" }
    - { amount: "1/2 tsp", item: "ground nutmeg" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 cup", item: "heavy cream or coconut milk" }
  steps:
    - "Heat the oven to 425°F. Toss the squash with 2 tbsp oil and roast 30 minutes until caramelized."
    - "Cook the onion in 1 tbsp oil in a large pot for 5 minutes. Add the garlic for 30 seconds."
    - "Add the squash, broth, nutmeg and salt and simmer 10 minutes."
    - "Blend until smooth, stir in the cream and heat through."
