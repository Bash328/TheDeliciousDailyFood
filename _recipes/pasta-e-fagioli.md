  ingredients:
    - { amount: "1 lb", item: "lean ground beef" }
    - { amount: "1 tbsp", item: "olive oil" }
    - { amount: "2", item: "medium carrots, diced small" }
    - { amount: "1", item: "large onion, diced small" }
    - { amount: "1", item: "celery stalk, diced small" }
    - { amount: "3", item: "garlic cloves, pressed" }
    - { amount: "15 oz", item: "canned diced tomatoes" }
    - { amount: "15 oz", item: "tomato sauce" }
    - { amount: "32 oz", item: "chicken broth" }
    - { amount: "15 oz", item: "canned Great Northern beans, drained and rinsed" }
    - { amount: "15 oz", item: "canned kidney beans, drained and rinsed" }
    - { amount: "1 cup", item: "ditalini pasta" }
    - { amount: "1 tbsp", item: "Italian seasoning" }
    - { amount: "1 tsp", item: "salt, to taste" }
    - { amount: "1 tsp", item: "black pepper, to taste" }
    - { amount: "2 tbsp", item: "fresh basil (optional)" }
  steps:
    - "Heat the oil in a large pot or Dutch oven over medium heat and brown the beef 3-5 minutes, breaking it up. Remove it and drain off the excess fat."
    - "In the same pot, cook the carrots, onion, celery and garlic about 4 minutes until softened."
    - "Add the diced tomatoes, tomato sauce and the beef back to the pot and stir."
    - "Add both beans and the broth and simmer about 10 minutes."
    - "Add the pasta, Italian seasoning, salt and pepper and cook another 10 minutes."
    - "Serve right away with basil or Parmesan."
note: "Adapted from Natasha's Kitchen's Olive Garden-style pasta e fagioli (4.99 stars from 319 votes). The pasta keeps absorbing broth, so thin leftovers with water."
