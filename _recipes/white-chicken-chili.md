recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 35
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "2 (15.5 oz) cans", item: "white beans, rinsed and drained" }
    - { amount: "4 cups", item: "low-sodium chicken broth, divided" }
    - { amount: "2 tbsp", item: "extra-virgin olive oil" }
    - { amount: "1 large", item: "yellow onion, diced" }
    - { amount: "1", item: "jalapeño, seeded and minced" }
    - { amount: "2 medium", item: "poblano peppers, seeded and diced" }
    - { amount: "4", item: "garlic cloves, minced" }
    - { amount: "1 tbsp", item: "ground cumin" }
    - { amount: "1 1/2 tsp", item: "ground coriander" }
    - { amount: "1 tsp", item: "ancho chili powder" }
    - { amount: "1/2 tsp", item: "salt, plus more to taste" }
    - { amount: "About 4 cups", item: "shredded rotisserie chicken (1 chicken, skin removed)" }
    - { amount: "3/4 cup", item: "frozen corn" }
    - { amount: "1 tbsp", item: "fresh lime juice, plus more to taste" }
    - { amount: "1/4 cup", item: "cilantro, chopped" }
    - { amount: "To serve", item: "sour cream, crushed tortilla chips, shredded cheddar and lime wedges" }
  steps:
    - "Blend one can of the beans with 1 cup of the broth until smooth. Keep the other can whole."
    - "Heat the oil in a large pot over medium-high heat and cook the onion, jalapeño and poblanos until soft, 4-5 minutes. Add the garlic for a minute."
    - "Stir in the cumin, coriander and chili powder for a minute to toast. Add the remaining 3 cups broth, the bean puree and the salt, bring to a boil, then simmer uncovered on low for 20 minutes."
    - "Add the chicken, whole beans, corn and lime juice and simmer about 5 minutes until hot."
    - "Season with more salt and lime, stir in the cilantro and serve with the toppings. It's meant to be soupy and thickens as it sits."
