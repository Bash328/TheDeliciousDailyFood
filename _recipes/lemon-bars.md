category: Dessert
cuisine: "American"
description: "A buttery shortbread crust under a bright, tangy lemon filling."
image: ""
recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 40
  servingsNumber: 16
  servingsUnit: Bars
  ingredients:
    - { amount: "1 cup", item: "butter, softened" }
    - { amount: "1/2 cup", item: "powdered sugar, plus more for dusting" }
    - { amount: "2 cups", item: "all-purpose flour" }
    - { amount: "4", item: "large eggs" }
    - { amount: "1 1/2 cups", item: "granulated sugar" }
    - { amount: "1/4 cup", item: "all-purpose flour, for the filling" }
    - { amount: "2 tsp", item: "lemon zest" }
    - { amount: "2/3 cup", item: "fresh lemon juice" }
  steps:
    - "Heat the oven to 350°F and line a 9x13-inch pan with parchment."
    - "Mix the butter, powdered sugar and 2 cups flour until crumbly. Press into the pan and bake 18-20 minutes until lightly golden."
    - "Whisk the eggs, granulated sugar, 1/4 cup flour, zest and lemon juice until smooth."
    - "Pour the filling over the warm crust and bake 20-25 minutes until set at the edges and just barely jiggly in the middle."
    - "Cool completely, dust with powdered sugar and cut into squares."
