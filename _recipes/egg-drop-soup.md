recipeContent:
  prepTimeMinutes: 5
  cookTimeMinutes: 10
  servingsNumber: 4
  servingsUnit: Bowls
  ingredients:
    - { amount: "4 cups", item: "chicken broth" }
    - { amount: "1 tsp", item: "fresh ginger, grated" }
    - { amount: "1 tbsp", item: "soy sauce" }
    - { amount: "2 tbsp", item: "cornstarch" }
    - { amount: "3 tbsp", item: "cold water" }
    - { amount: "2", item: "large eggs, beaten" }
    - { amount: "2", item: "scallions, sliced" }
    - { amount: "1/4 tsp", item: "white pepper" }
    - { amount: "1 tsp", item: "sesame oil" }
  steps:
    - "Bring the broth, ginger and soy sauce to a simmer."
    - "Mix the cornstarch with the cold water and stir into the broth until slightly thickened."
    - "Slowly pour the beaten eggs into the soup in a thin stream while stirring gently in one direction so they form ribbons."
    - "Remove from the heat and stir in the scallions, white pepper and sesame oil."
