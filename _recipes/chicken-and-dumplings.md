recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 35
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "3 tbsp", item: "butter" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "3", item: "carrots, sliced" }
    - { amount: "2", item: "celery stalks, sliced" }
    - { amount: "3 tbsp", item: "all-purpose flour" }
    - { amount: "6 cups", item: "chicken broth" }
    - { amount: "3 cups", item: "cooked chicken, shredded" }
    - { amount: "1 tsp", item: "dried thyme" }
    - { amount: "1/2 cup", item: "heavy cream" }
    - { amount: "2 cups", item: "all-purpose flour, for the dumplings" }
    - { amount: "1 tbsp", item: "baking powder" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "3/4 cup", item: "milk" }
    - { amount: "3 tbsp", item: "melted butter" }
  steps:
    - "Melt the butter and cook the onion, carrots and celery 6 minutes. Stir in 3 tbsp flour for 1 minute."
    - "Whisk in the broth, add the thyme and simmer 10 minutes. Stir in the chicken and cream."
    - "Whisk the 2 cups flour, baking powder and salt. Stir in the milk and melted butter to form a sticky dough."
    - "Drop spoonfuls of dough on top of the simmering soup, cover and cook 15 minutes without lifting the lid."
    - "The dumplings are done when they're puffed and cooked through."
note: "Keep the lid on while the dumplings cook so they steam and stay fluffy."
