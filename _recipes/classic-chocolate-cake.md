image: ""
recipeContent:
  prepTimeMinutes: 30
  cookTimeMinutes: 35
  servingsNumber: 12
  servingsUnit: Slices
  ingredients:
    - { amount: "1 3/4 cups", item: "all-purpose flour" }
    - { amount: "2 cups", item: "granulated sugar" }
    - { amount: "3/4 cup", item: "unsweetened cocoa powder" }
    - { amount: "2 tsp", item: "baking soda" }
    - { amount: "1 tsp", item: "baking powder" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "2", item: "large eggs" }
    - { amount: "1 cup", item: "buttermilk" }
    - { amount: "1 cup", item: "strong hot coffee" }
    - { amount: "1/2 cup", item: "vegetable oil" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "1 cup", item: "butter, softened" }
    - { amount: "3 cups", item: "powdered sugar" }
    - { amount: "1/2 cup", item: "cocoa powder, for the frosting" }
    - { amount: "1/3 cup", item: "milk" }
  steps:
    - "Heat the oven to 350°F. Grease and line two 9-inch round pans."
    - "Whisk the flour, sugar, cocoa, baking soda, baking powder and salt. Add the eggs, buttermilk, coffee, oil and vanilla and whisk until smooth. The batter will be thin."
    - "Divide between the pans and bake 30-35 minutes until a toothpick comes out clean. Cool completely."
    - "For the frosting, beat the butter until creamy, then add the powdered sugar, cocoa and milk and beat until fluffy."
    - "Frost between the layers and over the top and sides."
note: "The coffee deepens the chocolate flavor without tasting like coffee."
