  ingredients:
    - { amount: "2 cups", item: "all-purpose flour" }
    - { amount: "1 tbsp", item: "baking powder" }
    - { amount: "2 tbsp", item: "granulated sugar" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "2", item: "large eggs, separated" }
    - { amount: "1 3/4 cups", item: "milk" }
    - { amount: "1/2 cup", item: "butter, melted" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "To serve", item: "berries, whipped cream and maple syrup" }
  steps:
    - "Heat a waffle iron. Whisk the flour, baking powder, sugar and salt."
    - "Whisk the egg yolks, milk, melted butter and vanilla, then stir into the dry ingredients until just combined."
    - "Beat the egg whites to stiff peaks and gently fold them into the batter."
    - "Cook in the waffle iron according to its directions, usually 3-5 minutes, until golden and crisp."
    - "Serve right away with berries, whipped cream and syrup."
note: "Keep finished waffles in a 200°F oven on a wire rack so they stay crisp while you cook the rest."
