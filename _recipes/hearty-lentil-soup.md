    - { amount: "1", item: "onion, diced" }
    - { amount: "2", item: "carrots, diced" }
    - { amount: "2", item: "celery stalks, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 tsp", item: "ground cumin" }
    - { amount: "1/2 tsp", item: "turmeric" }
    - { amount: "1 1/2 cups", item: "brown or green lentils, rinsed" }
    - { amount: "1 can (14 oz)", item: "diced tomatoes" }
    - { amount: "6 cups", item: "vegetable broth" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "2 tbsp", item: "lemon juice" }
    - { amount: "2 cups", item: "fresh spinach" }
  steps:
    - "Cook the onion, carrots and celery in the oil for 6 minutes. Add the garlic, cumin and turmeric for 1 minute."
    - "Add the lentils, tomatoes, broth and salt and bring to a boil. Reduce the heat and simmer 30 minutes until the lentils are tender."
    - "Stir in the spinach until wilted and finish with lemon juice."
note: "A little lemon at the end brightens the earthy lentils."
