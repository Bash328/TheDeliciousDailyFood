recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 35
  servingsNumber: 8
  servingsUnit: Servings
  ingredients:
    - { amount: "9 oz", item: "pitted dates, roughly chopped" }
    - { amount: "1 tsp", item: "baking soda" }
    - { amount: "1 cup", item: "boiling water" }
    - { amount: "1/4 cup", item: "brown sugar, loosely packed" }
    - { amount: "6 tbsp", item: "unsalted butter, softened" }
    - { amount: "2", item: "eggs, room temperature" }
    - { amount: "1 1/4 cups", item: "all-purpose flour" }
    - { amount: "1 1/2 tsp", item: "baking powder" }
    - { amount: "1 cup", item: "brown sugar, tightly packed, for the sauce" }
    - { amount: "1 1/2 cups", item: "heavy cream" }
    - { amount: "1/2 tsp", item: "vanilla extract" }
    - { amount: "5 tbsp", item: "unsalted butter, for the sauce" }
    - { amount: "To serve", item: "vanilla ice cream or whipped cream" }
  steps:
    - "Heat the oven to 350°F. Grease and line a 8-inch square pan with an overhang (or grease individual pudding molds)."
    - "Put the dates in a bowl, sprinkle with the baking soda and pour over the boiling water. Let stand 10 minutes, then mash to a sloppy porridge."
    - "Beat the butter and 1/4 cup brown sugar until smooth, then beat in the eggs."
    - "Add the flour and sprinkle the baking powder over it, then mix just until combined. Quickly stir in the dates."
    - "Pour into the pan and bake 35 minutes until a skewer comes out clean (about 25 minutes for individual puddings)."
    - "For the sauce, heat the sugar, cream, vanilla and butter in a saucepan until melted, then simmer 2 minutes."
    - "While the pudding is hot, poke about 40 holes in it with a skewer and pour over 1/2 cup of the warm sauce. Let it soak 10 minutes."
    - "Lift out, cut and serve warm with the rest of the sauce and ice cream."
