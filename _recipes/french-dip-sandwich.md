    - { amount: "1 tsp", item: "garlic powder" }
    - { amount: "1 tsp", item: "dried thyme" }
    - { amount: "4", item: "hoagie or baguette rolls" }
    - { amount: "8 slices", item: "provolone or Swiss cheese" }
  steps:
    - "Simmer the broth, Worcestershire, garlic powder and thyme in a skillet for 5 minutes."
    - "Add the beef to the broth and warm through for 2 minutes."
    - "Split the rolls, top with the beef and cheese, and toast under the broiler until the cheese melts."
    - "Serve with small bowls of the warm broth for dipping."
