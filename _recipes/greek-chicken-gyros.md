    - { amount: "1 cup", item: "plain Greek yogurt (whole milk preferred)" }
    - { amount: "1", item: "lemon, juiced" }
    - { amount: "2 tbsp", item: "extra virgin olive oil" }
    - { amount: "2 tbsp", item: "red wine vinegar" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 tbsp", item: "oregano" }
    - { amount: "1 tsp", item: "sweet paprika" }
    - { amount: "1 tsp", item: "ground cumin" }
    - { amount: "1 tsp", item: "ground coriander" }
    - { amount: "1 pinch", item: "cayenne pepper (optional)" }
    - { amount: "To taste", item: "kosher salt and black pepper" }
    - { amount: "1 1/2 lbs", item: "chicken tenderloins" }
    - { amount: "1 tbsp", item: "olive oil, for the pan" }
    - { amount: "4", item: "pita breads" }
    - { amount: "To serve", item: "tzatziki sauce" }
    - { amount: "1", item: "large tomato, sliced" }
    - { amount: "1", item: "cucumber, sliced" }
    - { amount: "1", item: "green bell pepper, sliced" }
    - { amount: "1", item: "small red onion, sliced" }
    - { amount: "To serve", item: "pitted Kalamata olives" }
  steps:
    - "Whisk the yogurt, lemon juice, olive oil, vinegar, garlic, oregano, paprika, cumin, coriander, cayenne, salt and pepper in a large bowl. Add the chicken and coat well. Refrigerate 30 minutes or overnight."
    - "Heat 1 tbsp oil in a nonstick skillet over medium-high heat. Shake off excess marinade and cook the chicken undisturbed on one side about 5 minutes until well browned."
    - "Flip and cook about 5 minutes more until cooked through."
    - "Warm the pita over a gas flame or in a dry skillet."
    - "Spread each pita with tzatziki, add 2-3 chicken tenders and top with tomato, cucumber, pepper, onion and olives. Wrap and serve."
