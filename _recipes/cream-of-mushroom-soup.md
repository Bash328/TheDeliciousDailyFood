recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 30
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "4 tbsp", item: "butter" }
    - { amount: "1 1/2 lbs", item: "mushrooms, sliced" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1/4 cup", item: "all-purpose flour" }
    - { amount: "4 cups", item: "chicken or vegetable broth" }
    - { amount: "1 tsp", item: "fresh thyme leaves" }
    - { amount: "1 cup", item: "heavy cream" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
  steps:
    - "Melt the butter in a large pot. Cook the mushrooms in batches until deep golden, 8-10 minutes. Set a few aside for garnish."
    - "Cook the onion in the pot 5 minutes, then the garlic. Stir in the flour for 1 minute."
    - "Add the broth and thyme, return the mushrooms and simmer 15 minutes."
    - "Blend until smooth, stir in the cream and season. Top with the reserved mushrooms."
