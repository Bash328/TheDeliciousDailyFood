    - { amount: "1 lb", item: "flank steak, very thinly sliced against the grain" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "2 tbsp", item: "olive or vegetable oil, divided" }
    - { amount: "1 lb", item: "fresh broccoli, cut into florets (about 6 cups)" }
    - { amount: "1 tsp", item: "fresh ginger, grated" }
    - { amount: "2 tsp", item: "garlic, grated (3 cloves)" }
    - { amount: "1/2 cup", item: "warm low-sodium beef broth or water" }
    - { amount: "6 tbsp", item: "low-sodium soy sauce" }
    - { amount: "3 tbsp", item: "packed light brown sugar" }
    - { amount: "1 1/2 tbsp", item: "cornstarch" }
    - { amount: "2 tbsp", item: "sesame oil" }
    - { amount: "2 tsp", item: "sesame seeds (optional)" }
  steps:
    - "Start the rice first. Freeze the steak for 30 minutes so it slices thinly, then slice against the grain."
    - "Whisk the cornstarch with the soy sauce, brown sugar, sesame oil, ginger and garlic, then whisk in the warm broth. Whisk again just before using."
    - "Heat 1 tbsp oil in a large skillet over medium heat. Cook the broccoli 4-5 minutes with the lid partly on, tossing, until bright green and crisp-tender. Remove it."
    - "Turn the heat to high, add the remaining oil and cook the seasoned beef in a single layer, about 2 minutes per side."
    - "Add the sauce, lower the heat and simmer 3-4 minutes until thickened. Return the broccoli, toss, and thin with a splash of water if you'd like."
    - "Serve over rice, sprinkled with sesame seeds."
