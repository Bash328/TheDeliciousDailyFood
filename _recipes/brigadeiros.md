recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 10
  servingsNumber: 20
  servingsUnit: Truffles
  ingredients:
    - { amount: "1 can (14 oz)", item: "sweetened condensed milk" }
    - { amount: "3 tbsp", item: "unsweetened cocoa powder" }
    - { amount: "1 tsp", item: "butter, plus more for greasing" }
    - { amount: "To coat", item: "chocolate sprinkles (or dried coconut flakes)" }
  steps:
    - "Butter a plate and set it aside."
    - "Put the condensed milk, cocoa powder (sifted, if you like) and butter in a small pot."
    - "Cook over low heat 10-12 minutes, stirring constantly with a wooden spoon, until it almost sticks to the bottom. It's ready when you can run the spoon through the middle and the mixture takes 2-3 seconds to come back together."
    - "Turn off the heat, pour onto the buttered plate and chill 1 hour."
    - "Butter your hands, roll small spoonfuls into balls and coat in sprinkles. Set in little paper cases."
