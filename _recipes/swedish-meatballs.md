recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 25
  servingsNumber: 8
  servingsUnit: Servings
  ingredients:
    - { amount: "1 lb", item: "lean ground beef" }
    - { amount: "1 lb", item: "ground pork" }
    - { amount: "1 small", item: "onion, finely grated" }
    - { amount: "2", item: "garlic cloves, minced" }
    - { amount: "1 large", item: "egg" }
    - { amount: "1/3 cup", item: "panko breadcrumbs" }
    - { amount: "1/4 cup", item: "whole milk" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "1/4 tsp", item: "ground allspice" }
    - { amount: "1/4 tsp", item: "ground nutmeg" }
    - { amount: "2 tbsp", item: "oil, plus more as needed" }
    - { amount: "4 tbsp", item: "unsalted butter" }
    - { amount: "3 tbsp", item: "all-purpose flour" }
    - { amount: "3 cups", item: "beef broth" }
    - { amount: "1 cup", item: "heavy cream" }
    - { amount: "1/2 tsp", item: "Dijon mustard" }
    - { amount: "To serve", item: "mashed potatoes and lingonberry or cranberry jam" }
  steps:
    - "Soak the panko in the milk for a minute, then mix in the beef, pork, onion, garlic, egg, salt, pepper, allspice and nutmeg. Roll into 24 meatballs."
    - "Heat the oil in a large skillet over medium heat and brown the meatballs in two batches, turning, until cooked through, 6-7 minutes. Set aside, covered."
    - "Melt the butter in the same pan, whisk in the flour and cook until golden."
    - "Whisk in the broth, cream and mustard and simmer until the gravy thickens. Season with salt and pepper."
    - "Return the meatballs and simmer, turning them in the gravy, until hot through."
note: "Adapted from Natasha's Kitchen's Swedish meatballs (4.96 stars from 204 votes). Makes 24 meatballs."
