cuisine: "American"
description: "Dense, chocolatey brownies with a shiny crackly top."
image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 25
  servingsNumber: 16
  servingsUnit: Brownies
  ingredients:
    - { amount: "1/2 cup", item: "butter, melted" }
    - { amount: "1 cup", item: "granulated sugar" }
    - { amount: "2", item: "large eggs" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "1/3 cup", item: "unsweetened cocoa powder" }
    - { amount: "1/2 cup", item: "all-purpose flour" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "baking powder" }
    - { amount: "1/2 cup", item: "chocolate chips" }
  steps:
    - "Heat the oven to 350°F and line an 8-inch square pan with parchment."
    - "Whisk the melted butter and sugar together, then beat in the eggs and vanilla for 1 minute until glossy."
    - "Stir in the cocoa, flour, salt and baking powder until just combined, then fold in the chocolate chips."
    - "Spread in the pan and bake 20-25 minutes until a toothpick comes out with a few moist crumbs. Don't overbake."
    - "Cool completely before cutting for clean squares."
