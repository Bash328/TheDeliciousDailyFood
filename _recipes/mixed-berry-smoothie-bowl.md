recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 0
  servingsNumber: 2
  servingsUnit: Servings
  ingredients:
    - { amount: "2 cups", item: "frozen mixed berries (strawberries and blueberries)" }
    - { amount: "1", item: "ripe banana, plus more sliced for topping" }
    - { amount: "1/2 cup", item: "plain Greek yogurt" }
    - { amount: "1/4 cup", item: "milk, plus more as needed" }
    - { amount: "1 tbsp", item: "honey" }
    - { amount: "1/2 cup", item: "granola" }
    - { amount: "1/4 cup", item: "fresh blueberries" }
    - { amount: "1/4 cup", item: "fresh strawberries, halved" }
    - { amount: "1 tsp", item: "chia seeds" }
  steps:
    - "Add the frozen berries, banana, yogurt, milk, and honey to a blender. Blend until thick and smooth, adding a splash more milk only if it won't move."
    - "Divide the smoothie between two bowls."
    - "Arrange sliced banana, blueberries, and strawberries over the top."
    - "Finish with granola and a scatter of chia seeds. Serve immediately with a spoon."
note: "Keep the liquid to a minimum — the whole point is a bowl thick enough to hold its toppings."
