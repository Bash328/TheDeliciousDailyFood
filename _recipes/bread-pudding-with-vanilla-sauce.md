  prepTimeMinutes: 15

  servingsNumber: 9
  servingsUnit: Servings
  ingredients:
    - { amount: "6 cups", item: "day-old bread, cubed" }
    - { amount: "2 cups", item: "whole milk" }
    - { amount: "1 cup", item: "heavy cream" }
    - { amount: "4", item: "large eggs" }
    - { amount: "3/4 cup", item: "granulated sugar" }
    - { amount: "2 tsp", item: "vanilla extract" }
    - { amount: "1 tsp", item: "ground cinnamon" }
    - { amount: "1/2 cup", item: "raisins (optional)" }
    - { amount: "1/2 cup", item: "butter" }
    - { amount: "1/2 cup", item: "granulated sugar, for the sauce" }
    - { amount: "1/2 cup", item: "heavy cream, for the sauce" }
    - { amount: "1 tsp", item: "vanilla extract, for the sauce" }
  steps:
    - "Heat the oven to 350°F and grease a 9x9-inch baking dish. Add the bread cubes (and raisins, if using)."
    - "Whisk the milk, 1 cup cream, eggs, 3/4 cup sugar, vanilla and cinnamon. Pour over the bread and let soak 15 minutes."
    - "Bake 45-50 minutes until puffed, golden and the center is set."
    - "For the sauce, melt the butter in a small pan, stir in the sugar and cream and simmer 3 minutes. Stir in the vanilla."
    - "Spoon the warm sauce over each serving."
