date: 2026-01-01
category: Breakfast
cuisine: "American"
description: "Flaky buttermilk biscuits under creamy, peppery sausage gravy."
image: ""
recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 25
  servingsNumber: 6
  servingsUnit: Servings
  ingredients:
    - { amount: "2 1/2 cups", item: "all-purpose flour" }
    - { amount: "1 tbsp", item: "baking powder" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 cup", item: "cold butter, cubed" }
    - { amount: "1 cup", item: "cold buttermilk" }
    - { amount: "1 lb", item: "breakfast sausage" }
    - { amount: "1/4 cup", item: "all-purpose flour, for the gravy" }
    - { amount: "3 cups", item: "whole milk" }
    - { amount: "1 tsp", item: "black pepper" }
  steps:
    - "Heat the oven to 450°F. Whisk the flour, baking powder and salt. Cut in the butter until pea-sized, then stir in the buttermilk until just combined."
    - "Pat out to 1 inch thick, cut with a 3-inch cutter and place close together on a baking sheet."
    - "Bake 12-15 minutes until tall and golden."
    - "Brown the sausage in a skillet, breaking it up. Stir in the flour and cook 1 minute."
    - "Slowly pour in the milk and simmer, stirring, 5-8 minutes until thick. Season generously with pepper and salt."
    - "Split the biscuits and ladle the gravy over them."
