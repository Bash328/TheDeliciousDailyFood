image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 25
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "4 tbsp", item: "butter" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1/4 cup", item: "all-purpose flour" }
    - { amount: "2 cups", item: "chicken broth" }
    - { amount: "2 cups", item: "whole milk" }
    - { amount: "1 cup", item: "heavy cream" }
    - { amount: "4 cups", item: "broccoli florets, chopped" }
    - { amount: "2", item: "carrots, shredded" }
    - { amount: "3 cups", item: "sharp cheddar, shredded" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
  steps:
    - "Melt the butter in a pot and cook the onion 5 minutes. Add the garlic for 30 seconds."
    - "Whisk in the flour and cook 1 minute. Slowly whisk in the broth, milk and cream."
    - "Add the broccoli and carrots and simmer 15 minutes until tender."
    - "Remove from the heat and stir in the cheddar a handful at a time until melted. Season with salt and pepper."
    - "For a smoother soup, blend a portion with an immersion blender."
note: "Add the cheese off the heat so the soup doesn't turn grainy."
