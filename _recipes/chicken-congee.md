recipeContent:
  prepTimeMinutes: 30
  cookTimeMinutes: 50
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "1 cup", item: "Calrose or other medium-grain white rice" }
    - { amount: "8 cups", item: "unsalted chicken stock" }
    - { amount: "1 lb", item: "chicken breast, thinly sliced" }
    - { amount: "4 tsp", item: "oyster sauce" }
    - { amount: "4 tsp", item: "water" }
    - { amount: "2 tsp", item: "sesame oil" }
    - { amount: "2 tsp", item: "cornstarch" }
    - { amount: "1 tbsp", item: "fresh ginger, sliced into thin strips" }
    - { amount: "To taste", item: "salt and white pepper" }
    - { amount: "2 tbsp", item: "green onions, sliced, for serving" }
    - { amount: "2 tbsp", item: "cilantro, chopped, for serving" }
  steps:
    - "Rinse the rice in cool water, rubbing the grains, until the water runs clear. Soak 20 minutes, then drain."
    - "Toss the chicken slices with the oyster sauce, water, sesame oil and cornstarch and refrigerate."
    - "Bring the stock to a boil and stir in the rice. Cover until it returns to a boil, then reduce to medium-low with the lid ajar and simmer gently about 25 minutes, then stir. Keep simmering, stirring occasionally, until the rice has broken down into a creamy porridge, about 50 minutes in total."
    - "Raise the heat to medium and stir in the chicken a few pieces at a time. Cook 3-5 minutes until cooked through, add the ginger and cook 1 minute."
    - "Add more stock or water if too thick, and season with salt and white pepper."
    - "Ladle into bowls and top with green onions and cilantro."
