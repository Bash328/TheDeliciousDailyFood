recipeContent:
  prepTimeMinutes: 5
  cookTimeMinutes: 15
  servingsNumber: 10
  servingsUnit: Puffs
  ingredients:
    - { amount: "2 1/4 tsp", item: "active dry yeast (1 packet)" }
    - { amount: "1/2-3/4 cup", item: "sugar" }
    - { amount: "1/2 tbsp", item: "salt" }
    - { amount: "2 cups", item: "warm water (plus 1-2 tbsp for softer puffs)" }
    - { amount: "3 1/2 cups", item: "all-purpose flour" }
    - { amount: "For frying", item: "vegetable oil, at least 3 inches deep" }
    - { amount: "Optional", item: "cinnamon or powdered sugar, to roll" }
  steps:
    - "Mix the yeast, sugar, salt and warm water and let it sit 5 minutes to activate."
    - "Stir in the flour until smooth, cover and let it rise 1-2 hours."
    - "Heat at least 3 inches of oil in a large pot on low heat (too little oil gives flat puffs). Test with a drop of batter: it should rise to the top instead of sinking."
    - "Scoop small balls of batter with your hand or a spoon and drop them carefully into the oil."
    - "Fry a few minutes until the bottoms are golden, turn, and fry the other sides."
    - "Lift out onto napkins to drain. Roll in cinnamon or powdered sugar if you like."
note: "Adapted from Immaculate Bites' puff puff (4.87 stars from 135 votes). Best eaten warm."
