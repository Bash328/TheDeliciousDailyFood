  servingsNumber: 8
  servingsUnit: Bowls
  ingredients:
    - { amount: "1 1/2 lbs", item: "tomatillos, husks removed" }
    - { amount: "1/2 cup", item: "chopped onion" }
    - { amount: "1-2", item: "jalapeño or serrano chiles, stemmed" }
    - { amount: "3", item: "garlic cloves, smashed and peeled" }
    - { amount: "1 handful", item: "fresh cilantro leaves" }
    - { amount: "1/4 cup", item: "toasted pepitas" }
    - { amount: "1 tsp", item: "kosher salt" }
    - { amount: "1 carton (32 oz)", item: "chicken broth" }
    - { amount: "2 1/2 cups", item: "water" }
    - { amount: "2 lbs", item: "chicken breast" }
    - { amount: "1", item: "small white onion, quartered" }
    - { amount: "2", item: "garlic cloves, peeled" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "2 cans (29 oz each)", item: "Mexican-style hominy, drained" }
    - { amount: "To serve", item: "diced avocado, lime wedges, diced onion, sliced radishes, cilantro, shredded cabbage, Mexican oregano, crema, tostadas" }
  steps:
    - "Boil the tomatillos, onion, chiles and garlic in plenty of water about 8 minutes until the tomatillos turn light green, then drain."
    - "Blend them with the cilantro, pepitas and 1 tsp salt until smooth."
    - "Put the broth, water, chicken, quartered onion, 2 garlic cloves and 1 tsp salt in a large pot, bring to a boil, then simmer on medium-low about 20 minutes until the chicken is no longer pink."
    - "Shred the chicken. Blend the cooked onion and garlic from the poaching liquid into the green salsa until smooth."
    - "Return the shredded chicken to the broth with the salsa and the hominy. Partially cover, bring to a boil, then simmer 20 minutes. Season with salt if needed."
    - "Ladle into bowls and let everyone add their own garnishes. Serve with tostadas."
note: "Adapted from Muy Bueno's chicken pozole verde (4.93 stars from 27 votes)."
