    - { amount: "1 can (14 oz)", item: "diced tomatoes with green chilies" }
    - { amount: "1/2", item: "onion, diced" }
    - { amount: "2", item: "garlic cloves, minced" }
    - { amount: "1 tsp", item: "ground cumin" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "1 can (15 oz)", item: "refried or black beans, warmed" }
    - { amount: "8", item: "corn tortillas" }
    - { amount: "8", item: "eggs" }
    - { amount: "2 tbsp", item: "vegetable oil" }
    - { amount: "To serve", item: "avocado, cilantro and crumbled cotija cheese" }
  steps:
    - "Simmer the tomatoes, onion, garlic, cumin and salt 10 minutes into a thick salsa."
    - "Warm the tortillas in a dry skillet or crisp them in oil for 30 seconds per side."
    - "Fry the eggs in oil until the whites are set and yolks are runny."
    - "Spread beans on each tortilla, top with an egg and spoon over the salsa."
    - "Finish with avocado, cilantro and cotija."
