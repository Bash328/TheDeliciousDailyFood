image: ""
recipeContent:
  prepTimeMinutes: 5
  cookTimeMinutes: 40
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "8", item: "bone-in, skin-on chicken thighs" }
    - { amount: "2 tbsp", item: "olive oil" }
    - { amount: "1 tsp", item: "garlic powder" }
    - { amount: "1 tsp", item: "smoked paprika" }
    - { amount: "1 tsp", item: "dried thyme" }
    - { amount: "1 1/2 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
    - { amount: "1", item: "lemon, cut in wedges" }
  steps:
    - "Heat the oven to 425°F. Pat the chicken thoroughly dry with paper towels."
    - "Rub with the oil, then the garlic powder, paprika, thyme, salt and pepper."
    - "Place skin-side up on a rimmed baking sheet or in a cast-iron skillet."
    - "Bake 35-40 minutes until the skin is crisp and golden and the chicken reaches 175°F."
    - "Rest 5 minutes and serve with the lemon wedges."
note: "Dry skin is the secret to crispy skin. Don't skip the paper towels."
