    - { amount: "3", item: "carrots, diced" }
    - { amount: "3", item: "celery stalks, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 can (14 oz)", item: "diced tomatoes" }
    - { amount: "8 cups", item: "vegetable broth" }
    - { amount: "2", item: "potatoes, diced" }
    - { amount: "1 cup", item: "green beans, trimmed and chopped" }
    - { amount: "1 cup", item: "frozen peas or corn" }
    - { amount: "1 tsp", item: "dried Italian seasoning" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
  steps:
    - "Cook the onion, carrots and celery in the oil for 6 minutes. Add the garlic for 30 seconds."
    - "Add the tomatoes, broth, potatoes, green beans, seasoning, salt and pepper and bring to a boil."
    - "Simmer 20 minutes until the vegetables are tender. Add the peas or corn for the last 5 minutes."
    - "Taste and adjust the seasoning."
