image: ""
recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 50
  servingsNumber: 8
  servingsUnit: Servings
  ingredients:
    - { amount: "1 1/2 lbs", item: "chicken breast or thighs" }
    - { amount: "1 1/4 tsp", item: "salt, divided" }
    - { amount: "8", item: "flour tortillas (10-inch)" }
    - { amount: "3 oz", item: "cream cheese, room temperature" }
    - { amount: "1/3 cup", item: "sour cream" }
    - { amount: "4.5 oz", item: "diced green chilies, with their juice" }
    - { amount: "20 oz", item: "mild red enchilada sauce" }
    - { amount: "1/4 tsp", item: "ground black pepper" }
    - { amount: "1/2 tsp", item: "garlic powder" }
    - { amount: "1/2 tsp", item: "paprika" }
    - { amount: "2 1/2 cups", item: "shredded cheese (Mexican blend or Colby Jack), divided" }
  steps:
    - "Heat the oven to 400°F and grease a 9x13-inch dish."
    - "Cover the chicken with water in a saucepan, add 1 tsp salt and cook 8-10 minutes until done. Shred with two forks."
    - "Beat the cream cheese, then beat in the sour cream, green chilies, enchilada sauce, 1/4 tsp salt, pepper, garlic powder and paprika."
    - "Set a third of the sauce aside for the top. Mix the rest with the chicken and 2 cups of the cheese."
    - "Fill each tortilla with the chicken mixture, roll tightly and place seam-side down in the dish."
    - "Pour the reserved sauce over the top and sprinkle with the remaining cheese."
    - "Cover with foil and bake 30 minutes, then uncover and bake 10 minutes more until the cheese melts."
note: "Adapted from Natasha's Kitchen's chicken enchiladas (4.94 stars from 402 votes). Rotisserie chicken works well for the filling."
