recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 12
  servingsNumber: 24
  servingsUnit: Cookies
  ingredients:
    - { amount: "1 cup", item: "butter, softened" }
    - { amount: "1 cup", item: "packed brown sugar" }
    - { amount: "1/2 cup", item: "granulated sugar" }
    - { amount: "2", item: "large eggs" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "1 1/2 cups", item: "all-purpose flour" }
    - { amount: "1 tsp", item: "baking soda" }
    - { amount: "1 tsp", item: "ground cinnamon" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "3 cups", item: "old-fashioned rolled oats" }
    - { amount: "1 cup", item: "raisins" }
  steps:
    - "Heat the oven to 350°F. Beat the butter and both sugars until fluffy. Beat in the eggs and vanilla."
    - "Stir in the flour, baking soda, cinnamon and salt, then fold in the oats and raisins."
    - "Scoop rounded tablespoons onto baking sheets."
    - "Bake 10-12 minutes until the edges are golden and the centers look just set."
    - "Cool on the pan for 5 minutes, then move to a rack."
