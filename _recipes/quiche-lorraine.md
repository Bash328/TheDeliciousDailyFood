cuisine: "French"

image: ""
recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 45
  servingsNumber: 6
  servingsUnit: Slices
  ingredients:
    - { amount: "1", item: "refrigerated pie crust (9-inch)" }
    - { amount: "8 slices", item: "bacon, cooked and crumbled" }
    - { amount: "1 cup", item: "shredded Gruyère or Swiss" }
    - { amount: "4", item: "large eggs" }
    - { amount: "1 1/4 cups", item: "heavy cream" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "1/8 tsp", item: "ground nutmeg" }
  steps:
    - "Heat the oven to 400°F. Fit the crust into a 9-inch pie plate, prick the bottom with a fork, and bake 10-12 minutes. Reduce the oven to 350°F."
    - "Scatter the bacon and cheese over the crust."
    - "Whisk the eggs, cream, salt, pepper and nutmeg and pour over the filling."
    - "Bake 40-45 minutes until the center is just set and the top is golden."
    - "Cool 10 minutes before slicing."
