recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 75
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "4 tbsp", item: "butter" }
    - { amount: "3 lbs", item: "yellow onions, thinly sliced" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1 tsp", item: "sugar" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1/2 cup", item: "dry white wine" }
    - { amount: "6 cups", item: "beef broth" }
    - { amount: "2", item: "fresh thyme sprigs" }
    - { amount: "1", item: "bay leaf" }
    - { amount: "6 slices", item: "baguette, toasted" }
    - { amount: "2 cups", item: "shredded Gruyère" }
  steps:
    - "Melt the butter in a large heavy pot over medium heat. Add the onions, salt and sugar and cook, stirring often, 45-50 minutes until deep brown and jammy."
    - "Add the garlic for 1 minute, then pour in the wine and scrape up the browned bits."
    - "Add the broth, thyme and bay leaf and simmer 20 minutes. Remove the herbs."
    - "Heat the broiler. Ladle the soup into oven-safe bowls, top each with a baguette slice and cheese."
    - "Broil 2-3 minutes until bubbling and browned."
note: "Patience with the onions is the whole recipe. Don't rush the caramelizing."
