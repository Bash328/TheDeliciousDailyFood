  servingsUnit: Omelet
  ingredients:
    - { amount: "3", item: "large eggs" }
    - { amount: "1 tbsp", item: "milk" }
    - { amount: "Pinch", item: "salt and pepper" }
    - { amount: "1 tbsp", item: "butter" }
    - { amount: "1/3 cup", item: "shredded cheddar" }
    - { amount: "1 tbsp", item: "chives, chopped" }
  steps:
    - "Whisk the eggs, milk, salt and pepper until fully blended."
    - "Melt the butter in an 8-inch non-stick skillet over medium heat. Pour in the eggs."
    - "As the edges set, push them gently toward the center and tilt the pan so the uncooked egg flows underneath."
    - "When the top is almost set, sprinkle the cheese over one half and fold the other half over it."
    - "Slide onto a plate and top with chives."
note: "Cook low and gently. High heat makes omelets rubbery."
