recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 45
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "2", item: "large eggs" }
    - { amount: "2 tbsp", item: "vegetable oil or schmaltz" }
    - { amount: "1/2 cup", item: "matzo meal" }
    - { amount: "3 tbsp", item: "club soda or broth" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "8 cups", item: "chicken broth" }
    - { amount: "2", item: "carrots, sliced" }
    - { amount: "2", item: "celery stalks, sliced" }
    - { amount: "1 cup", item: "cooked chicken, shredded" }
    - { amount: "2 tbsp", item: "fresh dill" }
  steps:
    - "Whisk the eggs and oil, then stir in the matzo meal, club soda and salt. Chill 30 minutes."
    - "Wet your hands and roll the mixture into 12 balls."
    - "Boil the balls in salted water, covered, 30 minutes until fluffy."
    - "Meanwhile simmer the broth with the carrots and celery 15 minutes. Add the chicken."
    - "Transfer the matzo balls to the soup and serve with dill."
note: "Don't pack the balls too tightly. A light touch keeps them fluffy."
