    - { amount: "1", item: "cucumber, diced" }
    - { amount: "1/2 cup", item: "black olives, sliced" }
    - { amount: "1/2 cup", item: "pepperoni, diced" }
    - { amount: "1 cup", item: "mozzarella pearls" }
    - { amount: "1/2 cup", item: "Italian dressing, plus more as needed" }
    - { amount: "1/4 cup", item: "grated Parmesan" }
  steps:
    - "Cook the pasta in salted water until al dente. Drain and rinse under cold water until cool."
    - "Toss the pasta with the tomatoes, cucumber, olives, pepperoni and mozzarella."
    - "Pour in the dressing and Parmesan and stir to coat."
    - "Chill 30 minutes. Add a splash more dressing before serving, since the pasta absorbs it."
