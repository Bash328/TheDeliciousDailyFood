  ingredients:
    - { amount: "1/3 cup", item: "extra-virgin olive oil" }
    - { amount: "3 tbsp", item: "lemon juice" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "1/4 cup", item: "fine (#1) bulgur wheat" }
    - { amount: "3 bunches", item: "fresh curly parsley, finely chopped (about 3 packed cups)" }
    - { amount: "2", item: "firm tomatoes, finely chopped" }
    - { amount: "2", item: "green onions, finely chopped" }
    - { amount: "1/4 cup", item: "fresh mint, finely chopped" }
  steps:
    - "Whisk the oil, lemon juice, salt and pepper in a large bowl. Stir in the bulgur and let it soak up the dressing for 20-30 minutes until plump."
    - "Add the parsley, tomatoes, green onions and mint to one side of the bowl without mixing."
    - "When the bulgur is plump and most of the liquid is absorbed, toss everything together."
    - "Serve at room temperature or chilled."
