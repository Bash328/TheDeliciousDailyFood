    - { amount: "1/4 cup", item: "milk" }
    - { amount: "Pinch", item: "salt and pepper" }
    - { amount: "1 cup", item: "shredded cheese" }
    - { amount: "4", item: "large flour tortillas" }
    - { amount: "To serve", item: "salsa" }
  steps:
    - "Brown the sausage in a skillet and add the potatoes to crisp slightly. Remove and keep warm."
    - "Whisk the eggs, milk, salt and pepper, and scramble gently in the same pan until just set."
    - "Warm the tortillas and layer on the eggs, sausage and potatoes and cheese."
    - "Fold in the sides and roll tightly. Serve with salsa."
note: "Wrap in foil and freeze for grab-and-go breakfasts."
