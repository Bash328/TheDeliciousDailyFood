recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 15
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "1 1/2-2 cups (300 g)", item: "potatoes, diced" }
    - { amount: "1/2-3/4 cup (75 g)", item: "green peas" }
    - { amount: "1 cup (100 g)", item: "cauliflower florets (optional)" }
    - { amount: "1/2 cup (75 g)", item: "carrot, diced (optional)" }
    - { amount: "1 1/2 cups", item: "water, plus 1/2-3/4 cup more" }
    - { amount: "1 tbsp", item: "oil" }
    - { amount: "1 tbsp", item: "butter" }
    - { amount: "3/4-1 cup (125 g)", item: "onion, finely chopped" }
    - { amount: "1", item: "green chili, chopped (optional)" }
    - { amount: "1 tbsp", item: "ginger garlic paste" }
    - { amount: "1/2 cup", item: "green bell pepper, chopped" }
    - { amount: "3/4-1 cup (125 g)", item: "tomatoes, finely chopped" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1-1 1/2 tsp", item: "Kashmiri red chili powder" }
    - { amount: "1/4 tsp", item: "turmeric" }
    - { amount: "1-1 1/2 tbsp", item: "pav bhaji masala, plus 1/2 tsp for the buns" }
    - { amount: "3/4 tbsp", item: "kasuri methi (optional)" }
    - { amount: "8", item: "pav buns or soft dinner rolls" }
    - { amount: "3 tbsp", item: "butter, for toasting and topping" }
    - { amount: "To serve", item: "chopped onion, cilantro and lemon wedges" }
  steps:
    - "Boil the potatoes, peas, cauliflower and carrot in the water until very soft (or 2 whistles in a pressure cooker). Mash well."
    - "Heat the oil and butter in a wide pan over medium heat and cook the onion until translucent. Stir in the ginger garlic paste and chili until fragrant, then the bell pepper for 3-4 minutes."
    - "Add the tomatoes and salt and cook until soft and mushy. Stir in the chili powder, turmeric and pav bhaji masala and cook 2 minutes."
    - "Add the mashed vegetables and 1/2-3/4 cup water, and mash it all together over the heat for 3-5 minutes until thick. Add the kasuri methi and adjust the salt and spice."
    - "Split the buns, leaving one edge attached. Melt a little butter on a griddle, sprinkle with pav bhaji masala and toast the buns cut-side down for a minute or two."
    - "Serve the bhaji topped with a knob of butter, chopped onion and cilantro, with lemon to squeeze and the toasted buns for dipping."
note: "Adapted from Indian Healthy Recipes' pav bhaji (4.99 stars from 748 votes)."
