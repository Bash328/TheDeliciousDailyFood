ingredients:
  - { amount: "1 1/2 lbs", item: "pork shoulder, cut into 1-inch cubes" }
  - { amount: "3 tbsp", item: "olive oil" }
  - { amount: "2 tbsp", item: "lemon juice" }
  - { amount: "3", item: "garlic cloves, minced" }
  - { amount: "1 tbsp", item: "dried oregano" }
  - { amount: "1 tsp", item: "ground cumin" }
  - { amount: "1 tsp", item: "salt" }
  - { amount: "1/2 tsp", item: "black pepper" }
  - { amount: "4", item: "pita breads" }
  - { amount: "1 lb", item: "frozen fries, cooked per package" }
  - { amount: "1", item: "large tomato, sliced" }
  - { amount: "1/2", item: "red onion, thinly sliced" }
  - { amount: "2 tbsp", item: "fresh parsley, chopped" }
  - { amount: "1", item: "lemon, cut into wedges" }
  - { amount: "1 cup", item: "tzatziki, for serving" }
steps:
  - "Combine the pork with olive oil, lemon juice, garlic, oregano, cumin, salt, and pepper in a bowl. Toss to coat, cover, and marinate in the fridge at least 1 hour, up to overnight."
  - "Thread the pork onto skewers, soaking wooden skewers in water first if using."
  - "Heat a grill pan or outdoor grill to high. Grill the skewers, turning every few minutes, 10-12 minutes total until charred at the edges and cooked through."
  - "Meanwhile, cook the fries according to package directions, and warm the pita on the grill or in a dry skillet for a minute per side."
  - "Toss the red onion with the parsley and a squeeze of lemon."
  - "Divide the fries and warm pita among plates, lean the skewers alongside, and add the tomato slices and dressed onion. Serve with tzatziki and lemon wedges."
note: "For extra charred flavor, finish the skewers over an open flame or under a hot broiler for the last minute."
