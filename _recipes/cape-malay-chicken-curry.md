recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 120
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "2 lbs", item: "chicken pieces (thighs and drumsticks, skin on)" }
    - { amount: "2 tbsp", item: "vegetable oil" }
    - { amount: "9 oz", item: "white onion, sliced" }
    - { amount: "2 oz", item: "fresh ginger, grated" }
    - { amount: "3", item: "garlic cloves, crushed" }
    - { amount: "2 tsp", item: "ground cumin" }
    - { amount: "2 tsp", item: "ground coriander" }
    - { amount: "1 tsp", item: "ground turmeric" }
    - { amount: "2 tsp", item: "chili powder" }
    - { amount: "4", item: "cardamom pods, lightly crushed" }
    - { amount: "4", item: "whole cloves" }
    - { amount: "2", item: "cinnamon sticks" }
    - { amount: "1/2 tsp", item: "sugar" }
    - { amount: "1 tsp", item: "salt, plus more to taste" }
    - { amount: "1 1/2 cups", item: "chicken stock" }
    - { amount: "7 oz", item: "canned crushed tomatoes" }
    - { amount: "1 tbsp", item: "tomato paste" }
    - { amount: "2", item: "potatoes, peeled and cut into chunks" }
  steps:
    - "Heat the oil in a Dutch oven over medium heat and fry the onions until they start to turn golden. Stir in the salt and sugar."
    - "Add the ginger and, once fragrant, the chicken. Cook until the chicken has some color, then add the garlic for a minute."
    - "Stir in the cumin, coriander, turmeric, chili powder, cardamom, cloves and cinnamon until fragrant."
    - "Pour in the stock, cover and cook on a slow bubble for 40 minutes, adding a splash of water if it reduces too far."
    - "Add the tomatoes and tomato paste and cook uncovered for 30 minutes."
    - "Meanwhile boil the potatoes in salted water until tender, then add them to the curry. Taste and adjust the seasoning."
    - "Serve now, or cool and refrigerate overnight, which traditionally deepens the flavor. Reheat and serve with yellow rice."
note: "Adapted from Grantourismo's Cape Malay chicken curry (5 stars from 5 votes, a small sample)."
