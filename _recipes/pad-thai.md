  servingsUnit: Servings
  ingredients:
    - { amount: "8 oz", item: "flat rice noodles" }
    - { amount: "3 tbsp", item: "oil, divided" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "8 oz", item: "shrimp, chicken or extra-firm tofu, cut into small pieces" }
    - { amount: "2", item: "eggs" }
    - { amount: "1 cup", item: "fresh bean sprouts" }
    - { amount: "1", item: "red bell pepper, thinly sliced" }
    - { amount: "3", item: "green onions, chopped" }
    - { amount: "1/2 cup", item: "dry roasted peanuts" }
    - { amount: "2", item: "limes" }
    - { amount: "1/2 cup", item: "fresh cilantro, chopped" }
    - { amount: "3 tbsp", item: "fish sauce" }
    - { amount: "1 tbsp", item: "low-sodium soy sauce" }
    - { amount: "5 tbsp", item: "light brown sugar" }
    - { amount: "2 tbsp", item: "rice vinegar or tamarind paste" }
    - { amount: "1 tbsp", item: "Sriracha, or more to taste" }
    - { amount: "2 tbsp", item: "creamy peanut butter (optional)" }
  steps:
    - "Cook the noodles just until tender, then rinse under cold water."
    - "Whisk the fish sauce, soy sauce, sugar, vinegar, Sriracha and peanut butter in a bowl."
    - "Heat about 1 1/2 tbsp oil in a wok over medium-high heat and cook the shrimp, chicken or tofu with the garlic and bell pepper. Shrimp takes 1-2 minutes per side, chicken 3-4 minutes."
    - "Push everything aside, add a little oil and scramble the eggs in small pieces."
    - "Add the noodles, sauce, bean sprouts and most of the peanuts and toss to combine."
    - "Top with green onions, remaining peanuts, cilantro and lime wedges, and serve right away."
note: "Adapted from Tastes Better From Scratch's pad Thai (4.97 stars from 5,184 votes). It uses rice vinegar for a quick sour note; tamarind paste is more traditional."
