recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 25
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "1 tbsp", item: "vegetable oil" }
    - { amount: "1 lb", item: "boneless chicken thighs, cut into pieces" }
    - { amount: "", item: "Salt and black pepper" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 tbsp", item: "fresh ginger, grated" }
    - { amount: "3 tbsp", item: "red curry paste" }
    - { amount: "1 can (14 oz)", item: "full-fat coconut milk" }
    - { amount: "2 cups", item: "chicken broth" }
    - { amount: "6 oz", item: "egg noodles" }
    - { amount: "1 cup", item: "cherry tomatoes, halved" }
    - { amount: "1 cup", item: "snap peas, trimmed" }
    - { amount: "1", item: "lime, juiced, plus wedges for serving" }
    - { amount: "", item: "Fresh cilantro, for serving" }
  steps:
    - "Season the chicken with salt and pepper. Heat oil in a large pot over medium-high heat and sear the chicken 4-5 minutes, until browned. Remove and set aside."
    - "Lower the heat and add garlic, ginger, and curry paste to the pot, stirring for a minute until fragrant."
    - "Pour in the coconut milk and chicken broth, scraping up anything stuck to the bottom. Bring to a simmer and return the chicken to the pot, cooking 10 minutes until cooked through."
    - "Add the egg noodles and simmer 4-5 minutes until just tender. Stir in the cherry tomatoes and snap peas and cook 2 minutes more, until the tomatoes soften and the peas are bright green."
    - "Stir in the lime juice, then ladle into bowls and top with cilantro and lime wedges."
note: "Snap peas go in last so they keep their crunch — don't let the soup simmer long after adding them."
