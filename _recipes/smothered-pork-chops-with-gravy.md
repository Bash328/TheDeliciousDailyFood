date: 2026-01-01
category: Dinner
cuisine: "American"
description: "Pan-seared pork chops simmered in a rich onion gravy."
image: ""
recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 30
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "4", item: "bone-in pork chops, 3/4-inch thick" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
    - { amount: "1/4 cup", item: "all-purpose flour, plus 2 tbsp for the gravy" }
    - { amount: "3 tbsp", item: "vegetable oil" }
    - { amount: "1", item: "large onion, sliced" }
    - { amount: "2 cups", item: "chicken broth" }
    - { amount: "1 tsp", item: "dried thyme" }
    - { amount: "1 tsp", item: "garlic powder" }
  steps:
    - "Season the chops with salt and pepper and dredge in 1/4 cup flour."
    - "Brown in the oil in a large skillet over medium-high heat, 3 minutes per side. Remove to a plate."
    - "Cook the onion in the pan 6 minutes until golden. Sprinkle in 2 tbsp flour and stir 1 minute."
    - "Whisk in the broth, thyme and garlic powder and bring to a simmer."
    - "Return the chops to the pan, cover and simmer 12-15 minutes until the pork reaches 145°F."
    - "Serve with the gravy over rice or mashed potatoes."
