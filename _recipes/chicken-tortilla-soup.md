    - { amount: "2 tbsp", item: "olive oil" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1", item: "jalapeño, seeded and diced" }
    - { amount: "2 tsp", item: "chili powder" }
    - { amount: "1 tsp", item: "ground cumin" }
    - { amount: "1 can (14 oz)", item: "fire-roasted diced tomatoes" }
    - { amount: "6 cups", item: "chicken broth" }
    - { amount: "2", item: "chicken breasts" }
    - { amount: "1 can (15 oz)", item: "black beans, drained" }
    - { amount: "1 cup", item: "corn" }
    - { amount: "2 tbsp", item: "lime juice" }
    - { amount: "To serve", item: "tortilla strips, avocado, cilantro and cheese" }
  steps:
    - "Cook the onion and jalapeño in the oil 5 minutes. Add the garlic, chili powder and cumin for 1 minute."
    - "Add the tomatoes, broth and chicken and simmer 20 minutes until the chicken is cooked through."
    - "Remove the chicken, shred it, and return to the pot with the beans and corn. Simmer 5 minutes."
    - "Finish with the lime juice and serve with tortilla strips, avocado, cilantro and cheese."
