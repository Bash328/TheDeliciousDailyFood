recipeContent:
  prepTimeMinutes: 5
  cookTimeMinutes: 10
  servingsNumber: 2
  servingsUnit: Servings
  ingredients:
    - { amount: "2 tbsp", item: "oil, plus more for the egg" }
    - { amount: "1/2", item: "onion, finely chopped" }
    - { amount: "2", item: "garlic cloves, minced" }
    - { amount: "4 oz", item: "bacon or ham, chopped (optional)" }
    - { amount: "1 to 1 1/2 cups", item: "diced vegetables, fresh or frozen" }
    - { amount: "2 cups", item: "cooked day-old white rice, packed" }
    - { amount: "3", item: "eggs, whisked" }
    - { amount: "3/4 cup", item: "green onions, sliced" }
    - { amount: "1 tbsp", item: "Chinese cooking wine, mirin or dry sherry" }
    - { amount: "1 tbsp", item: "oyster sauce" }
    - { amount: "1 tbsp", item: "light soy sauce" }
    - { amount: "1 tsp", item: "sesame oil (optional)" }
    - { amount: "1/4 tsp", item: "white pepper" }
  steps:
    - "Heat the oil in a wok or large skillet over high heat until smoking."
    - "Cook the onion, garlic and bacon about 1 1/2 minutes until the bacon is light golden."
    - "Add the vegetables (frozen is fine) and cook 2 minutes."
    - "Add the rice, wine, oyster sauce, soy sauce, sesame oil and white pepper and stir-fry about 1 1/2 minutes until the liquid has cooked off."
    - "Push the rice to the side, add a little oil to the cleared space and scramble the eggs fully there."
    - "Stir in the green onions, mix the egg through the rice and serve straight away."
note: "Adapted from RecipeTin Eats' fried rice (4.99 stars from 302 votes). Day-old rice matters: freshly cooked rice turns mushy."
