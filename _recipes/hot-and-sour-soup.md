    - { amount: "6 cups", item: "beef broth" }
    - { amount: "1/2 cup", item: "mushrooms, sliced" }
    - { amount: "1/4 cup", item: "soy sauce" }
    - { amount: "1/4 cup", item: "rice vinegar" }
    - { amount: "1 tbsp", item: "sriracha" }
    - { amount: "1 tsp", item: "white pepper" }
    - { amount: "3 tbsp", item: "cornstarch" }
    - { amount: "3 tbsp", item: "cold water" }
    - { amount: "2", item: "large eggs, well beaten" }
    - { amount: "6 oz", item: "firm tofu, cut into thin strips" }
    - { amount: "2", item: "green onions, sliced" }
  steps:
    - "Bring the broth to a simmer in a large pot. Add the mushrooms, soy sauce, vinegar, sriracha and white pepper and simmer 5 minutes."
    - "Whisk the cornstarch with the cold water and stir it in. Simmer about 5 minutes until the soup thickens."
    - "Slowly pour in the beaten eggs while stirring the soup in a circle so they form ribbons."
    - "Stir in the tofu and green onions, take off the heat and serve right away."
