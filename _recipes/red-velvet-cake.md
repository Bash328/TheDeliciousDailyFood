recipeContent:
  prepTimeMinutes: 30
  cookTimeMinutes: 30
  servingsNumber: 12
  servingsUnit: Slices
  ingredients:
    - { amount: "2 1/2 cups", item: "all-purpose flour" }
    - { amount: "1 1/2 cups", item: "granulated sugar" }
    - { amount: "1 tsp", item: "baking soda" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1 tbsp", item: "unsweetened cocoa powder" }
    - { amount: "1 1/2 cups", item: "vegetable oil" }
    - { amount: "1 cup", item: "buttermilk" }
    - { amount: "2", item: "large eggs" }
    - { amount: "2 tbsp", item: "red food coloring" }
    - { amount: "1 tsp", item: "white vinegar" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "16 oz", item: "cream cheese, softened" }
    - { amount: "1 cup", item: "butter, softened" }
    - { amount: "4 cups", item: "powdered sugar" }
  steps:
    - "Heat the oven to 350°F and grease and line two 9-inch round pans."
    - "Whisk the flour, sugar, baking soda, salt and cocoa."
    - "Whisk in the oil, buttermilk, eggs, food coloring, vinegar and vanilla until smooth."
    - "Divide between the pans and bake 28-32 minutes until a toothpick comes out clean. Cool completely."
    - "Beat the cream cheese and butter until smooth, then beat in the powdered sugar. Frost the cake."
