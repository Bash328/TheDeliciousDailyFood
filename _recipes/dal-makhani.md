  prepTimeMinutes: 10

  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "1 cup (200 g)", item: "whole urad dal (black lentils)" }
    - { amount: "1/4 cup (55 g)", item: "rajma (red kidney beans)" }
    - { amount: "4 cups", item: "water, for cooking the dal" }
    - { amount: "2 tbsp", item: "ghee" }
    - { amount: "1", item: "bay leaf" }
    - { amount: "2", item: "green cardamom pods" }
    - { amount: "1", item: "black cardamom pod" }
    - { amount: "1/2 cup", item: "finely chopped onion" }
    - { amount: "1 tbsp", item: "ginger-garlic paste" }
    - { amount: "1 cup (220 g)", item: "tomato puree" }
    - { amount: "3/4 to 1 1/2 tsp", item: "Kashmiri red chili powder" }
    - { amount: "1 tsp", item: "garam masala, plus more to taste" }
    - { amount: "1 tsp", item: "sea salt" }
    - { amount: "2 cups", item: "hot water, added later" }
    - { amount: "1 tbsp", item: "kasuri methi (dried fenugreek leaves)" }
    - { amount: "1/4 to 1/3 cup", item: "heavy cream" }
    - { amount: "3 tbsp", item: "unsalted butter" }
    - { amount: "To serve", item: "basmati rice or naan" }
  steps:
    - "Rinse the urad dal and rajma three times, then soak in 4 cups of water overnight. Drain and rinse again."
    - "Pressure cook with 4 cups of fresh water for 10 whistles (or 20 minutes at high pressure in an Instant Pot) until both are very soft and mash easily. Without a pressure cooker, simmer in a heavy pot, adding hot water as needed."
    - "Heat the ghee in a heavy pot with the bay leaf and cardamoms until they sizzle. Cook the onion about 7 minutes until golden, then the ginger-garlic paste for 1 minute."
    - "Add the tomato puree and cook 3-4 minutes. Stir in the salt, chili powder and garam masala and cook 4-5 minutes until thick."
    - "Mash the cooked dal lightly and add it with its cooking liquid and 1 cup of hot water."
    - "Cook on low about 60 minutes, stirring often, until thickened. Add another 1/2 to 1 cup hot water and cook about 30 minutes more until really thick and creamy (about 90 minutes total)."
    - "Crush in the kasuri methi, stir in the cream and cook 6-8 minutes. Taste and add salt or garam masala, then stir in the butter."
    - "Serve with rice or naan, with a swirl of cream if you like."
