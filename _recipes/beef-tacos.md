    - { amount: "1 tsp", item: "ground cumin" }
    - { amount: "1 tsp", item: "paprika" }
    - { amount: "1/2 tsp", item: "garlic powder" }
    - { amount: "1/2 tsp", item: "salt" }

    - { amount: "To serve", item: "sour cream and salsa" }
  steps:
    - "Brown the beef with the onion in a skillet over medium-high heat, 7 minutes. Drain off excess fat."
    - "Stir in the chili powder, cumin, paprika, garlic powder and salt, then add the water."
    - "Simmer 5 minutes until the liquid is mostly absorbed."
    - "Warm the tortillas in a dry pan or the oven."
    - "Fill with the beef and top with lettuce, cheese, tomato, sour cream and salsa."
