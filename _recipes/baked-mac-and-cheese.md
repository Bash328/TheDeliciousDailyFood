cuisine: "American"
description: "Creamy cheddar sauce, tender pasta and a golden breadcrumb top."
image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 30
  servingsNumber: 8
  servingsUnit: Servings
  ingredients:
    - { amount: "1 lb", item: "elbow macaroni" }
    - { amount: "4 tbsp", item: "butter" }
    - { amount: "1/4 cup", item: "all-purpose flour" }
    - { amount: "3 cups", item: "whole milk" }
    - { amount: "1/2 tsp", item: "mustard powder" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "4 cups", item: "sharp cheddar, shredded" }
    - { amount: "1 cup", item: "Gruyère or Monterey Jack, shredded" }
    - { amount: "1 cup", item: "panko breadcrumbs" }
    - { amount: "2 tbsp", item: "butter, melted" }
  steps:
    - "Heat the oven to 375°F and butter a 9x13-inch dish. Boil the pasta 2 minutes short of the package time, then drain."
    - "Melt the butter in a large pot over medium heat. Whisk in the flour for 1 minute."
    - "Slowly whisk in the milk and cook, stirring, 5 minutes until thickened. Stir in the mustard powder, salt and pepper."
    - "Remove from the heat and stir in the cheeses until melted and smooth. Fold in the pasta."
    - "Pour into the dish. Mix the panko with the melted butter and sprinkle over the top."
    - "Bake 25-30 minutes until bubbly and golden."
note: "Shred the cheese yourself. Pre-shredded cheese has coatings that can make the sauce grainy."
