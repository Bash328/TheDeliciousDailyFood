image: ""
recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 10
  servingsNumber: 24
  servingsUnit: Cookies
  ingredients:
    - { amount: "1 cup", item: "creamy peanut butter" }
    - { amount: "1/2 cup", item: "butter, softened" }
    - { amount: "1/2 cup", item: "granulated sugar" }
    - { amount: "1/2 cup", item: "packed brown sugar" }
    - { amount: "1", item: "large egg" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "1 1/4 cups", item: "all-purpose flour" }
    - { amount: "1/2 tsp", item: "baking soda" }
    - { amount: "1/4 tsp", item: "salt" }
  steps:
    - "Heat the oven to 350°F. Beat the peanut butter, butter and both sugars until fluffy. Beat in the egg and vanilla."
    - "Stir in the flour, baking soda and salt until combined."
    - "Roll the dough into 1-inch balls, set on baking sheets and press with a fork in a crisscross pattern."
    - "Bake 9-10 minutes until the edges are just set."
    - "Cool on the pan for 5 minutes before moving."
