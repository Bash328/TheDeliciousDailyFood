date: 2026-01-01
category: Lunch
cuisine: "American"
description: "Crisp romaine, homemade croutons and a punchy garlic-anchovy dressing."
image: ""
recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 10
  servingsNumber: 4
  servingsUnit: Salads
  ingredients:
    - { amount: "2 heads", item: "romaine lettuce, chopped" }
    - { amount: "2 cups", item: "bread cubes" }
    - { amount: "2 tbsp", item: "olive oil" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "2", item: "garlic cloves, minced" }
    - { amount: "2", item: "anchovy fillets, minced" }
    - { amount: "1", item: "egg yolk" }
    - { amount: "1 tbsp", item: "lemon juice" }
    - { amount: "1 tsp", item: "Dijon mustard" }
    - { amount: "1 tsp", item: "Worcestershire sauce" }
    - { amount: "1/2 cup", item: "olive oil, for the dressing" }
    - { amount: "3/4 cup", item: "grated Parmesan" }
  steps:
    - "Toss the bread cubes with 2 tbsp oil and the salt and bake at 375°F for 10 minutes until golden, turning once."
    - "Mash the garlic and anchovies to a paste in a large bowl. Whisk in the yolk, lemon juice, mustard and Worcestershire."
    - "Drizzle in the 1/2 cup of oil slowly while whisking until thick and creamy."
    - "Add the romaine and toss to coat, then fold in the Parmesan and croutons."
    - "Serve right away with extra black pepper."
note: "The egg yolk is traditional. For a no-raw-egg version use 2 tbsp mayonnaise instead."
