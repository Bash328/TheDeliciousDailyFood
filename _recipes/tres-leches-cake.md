recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 25
  servingsNumber: 16
  servingsUnit: Slices
  ingredients:
    - { amount: "1 1/2 cups", item: "all-purpose flour" }
    - { amount: "2 tsp", item: "baking powder" }
    - { amount: "3/4 tsp", item: "kosher salt" }
    - { amount: "3", item: "large eggs, room temperature" }
    - { amount: "1 cup", item: "white sugar" }
    - { amount: "2 tsp", item: "vanilla extract" }
    - { amount: "1/2 cup", item: "full-fat milk, room temperature" }
    - { amount: "1 can (12 oz)", item: "evaporated milk" }
    - { amount: "1 can (14 oz)", item: "sweetened condensed milk" }
    - { amount: "1 cup", item: "heavy cream" }
    - { amount: "1 tsp", item: "vanilla extract, for the milk mix" }
    - { amount: "1/2 tsp", item: "almond extract (optional)" }
    - { amount: "1 1/2 cups", item: "heavy cream, for the topping" }
    - { amount: "3 tbsp", item: "white sugar" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "1/2 tsp+", item: "cinnamon, for sprinkling" }
  steps:
    - "Heat the oven to 350°F and butter a 9x13-inch metal pan (not glass or ceramic, and don't line it)."
    - "Whisk the flour, baking powder and salt."
    - "Beat the eggs, sugar and vanilla on medium-high for 10 minutes until pale yellow and fluffy."
    - "On low speed, add half the flour, then the milk, then the rest of the flour, stopping as soon as most of the flour is mixed in. Finish folding by hand with a spatula, scraping the bottom, without deflating the batter."
    - "Pour into the pan and bake 25 minutes, until a skewer comes out clean. Cool 30 minutes."
    - "Whisk the evaporated milk, condensed milk, 1 cup cream, vanilla and almond extract together."
    - "Poke lots of holes all the way to the bottom of the cake with a thin chopstick. Pour the milk mixture over in about 3 additions, a minute or two apart, so it soaks in instead of pooling."
    - "Refrigerate at least 4 hours or overnight."
    - "Whip the topping cream with the sugar and vanilla until firm, spread over the cake and dust with cinnamon. Serve from the pan."
note: "Adapted from RecipeTin Eats' tres leches cake (4.88 stars from 39 votes). It's even better the next day."
