recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 30
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "4 slices", item: "bacon, chopped" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "2", item: "celery stalks, diced" }
    - { amount: "2 tbsp", item: "all-purpose flour" }
    - { amount: "3 cups", item: "chicken broth" }
    - { amount: "1 lb", item: "potatoes, diced" }
    - { amount: "4 cups", item: "corn kernels (fresh or frozen)" }
    - { amount: "1 1/2 cups", item: "milk" }
    - { amount: "1/2 cup", item: "heavy cream" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
  steps:
    - "Cook the bacon in a pot until crisp. Remove, leaving 2 tbsp fat."
    - "Cook the onion and celery in the fat 5 minutes. Stir in the flour for 1 minute."
    - "Add the broth and potatoes and simmer 12 minutes until tender."
    - "Add the corn, milk, cream, salt and pepper and simmer 8 minutes. Mash some of the potatoes against the side to thicken."
    - "Serve topped with the bacon."
