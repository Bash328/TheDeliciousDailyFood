    - { amount: "1", item: "medium carrot, diced" }
    - { amount: "1 tsp", item: "sea salt" }
    - { amount: "To taste", item: "freshly ground black pepper" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 tsp", item: "ground cumin" }
    - { amount: "1 tsp", item: "ground coriander" }
    - { amount: "1/2 tsp", item: "chili powder" }
    - { amount: "2 cans (15 oz each)", item: "black beans, including the liquid" }

    - { amount: "1 tbsp", item: "fresh lime juice, plus wedges" }
    - { amount: "To serve", item: "avocado, cilantro, sour cream, pickled onions" }
  steps:
    - "Heat the oil in a large pot over medium heat. Cook the onion, celery, carrot, salt and pepper 10-15 minutes until soft, stirring occasionally."
    - "Add the garlic, cumin, coriander and chili powder and stir 30 seconds."
    - "Add the beans with their liquid, the chipotles, adobo sauce and broth. Simmer 30 minutes."
    - "Cool slightly, then puree half the soup in a blender (or use an immersion blender in the pot), and stir it back in. Stir in the lime juice."
    - "Season to taste and serve with lime wedges and your favorite toppings."
