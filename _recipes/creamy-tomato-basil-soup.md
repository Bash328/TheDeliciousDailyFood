    - { amount: "2 cups", item: "vegetable or chicken broth" }
    - { amount: "1 tsp", item: "sugar" }

    - { amount: "1/2 cup", item: "fresh basil leaves" }
    - { amount: "1/2 cup", item: "heavy cream" }
  steps:
    - "Cook the onion in the oil for 5 minutes. Add the garlic for 30 seconds."
    - "Add the tomatoes with their juices, broth, sugar and salt and simmer 20 minutes."
    - "Add the basil and blend until smooth with an immersion blender."
    - "Stir in the cream and heat through without boiling."
    - "Serve with grilled cheese for dipping."
