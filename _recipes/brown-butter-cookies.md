ingredients:
  - { amount: "1 cup", item: "unsalted butter" }
  - { amount: "1 cup", item: "brown sugar, packed" }
  - { amount: "1/2 cup", item: "granulated sugar" }
  - { amount: "2", item: "large eggs" }
  - { amount: "1 tsp", item: "vanilla extract" }
  - { amount: "2 1/4 cups", item: "all-purpose flour" }
  - { amount: "1 tsp", item: "baking soda" }
  - { amount: "1 tsp", item: "flaky sea salt, plus more for topping" }
  - { amount: "2 cups", item: "chocolate chips, mixed sizes" }
steps:
  - "Brown the butter in a light-colored saucepan over medium heat, swirling often, until it smells nutty and turns amber. Pour into a bowl to cool for 15 minutes."
  - "Whisk the cooled brown butter with both sugars until glossy. Beat in the eggs and vanilla."
  - "Fold in flour, baking soda, and salt just until no dry streaks remain. Fold in the chocolate chips."
  - "Chill the dough for at least 30 minutes — this keeps the cookies thick. Preheat oven to 350°F (175°C)."
  - "Scoop 2-tablespoon balls onto a lined sheet, spaced well apart. Top with flaky salt."
  - "Bake 10-12 minutes until the edges are set but centers look slightly underdone. Cool on the sheet for 5 minutes before moving — they finish setting there."
note: "Dough freezes well pre-scooped — bake straight from frozen, add 2 extra minutes."
