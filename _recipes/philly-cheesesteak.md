image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 15
  servingsNumber: 4
  servingsUnit: Sandwiches
  ingredients:
    - { amount: "1 1/2 lbs", item: "ribeye or sirloin, thinly sliced" }
    - { amount: "2 tbsp", item: "vegetable oil" }
    - { amount: "1", item: "large onion, sliced" }
    - { amount: "1", item: "green bell pepper, sliced" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/2 tsp", item: "black pepper" }
    - { amount: "8 slices", item: "provolone cheese" }
    - { amount: "4", item: "hoagie rolls, split" }
  steps:
    - "Freeze the steak 20 minutes so it's easier to slice very thin."
    - "Cook the onion and pepper in 1 tbsp oil in a large skillet over medium-high heat 6-8 minutes until softened. Remove."
    - "Add the remaining oil and sear the steak 2-3 minutes, breaking it up, seasoning with salt and pepper."
    - "Return the vegetables to the pan and top the meat with the provolone. Cover for a minute to melt."
    - "Pile into the rolls."
note: "Slicing the steak against the grain keeps it tender."
