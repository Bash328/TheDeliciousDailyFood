image: ""
recipeContent:
  prepTimeMinutes: 15
  cookTimeMinutes: 25
  servingsNumber: 4
  servingsUnit: Servings
  ingredients:
    - { amount: "12 oz", item: "fresh spinach (about 350g)" }
    - { amount: "1", item: "medium tomato" }
    - { amount: "5", item: "large garlic cloves (3 whole, 2 finely chopped)" }
    - { amount: "1 inch", item: "fresh ginger" }
    - { amount: "1", item: "green chili, or to taste" }
    - { amount: "3", item: "raw cashews, soaked 5 minutes in hot water (optional)" }
    - { amount: "1 1/2 tbsp", item: "oil" }
    - { amount: "1", item: "large onion, finely chopped" }
    - { amount: "1/2 cup", item: "water, or as needed" }
    - { amount: "3/4-1 tsp", item: "garam masala" }
    - { amount: "1/4 tsp", item: "turmeric powder" }
    - { amount: "1/4 tsp", item: "red chili powder" }
    - { amount: "3/4 tsp", item: "salt" }
    - { amount: "2-3 tbsp", item: "heavy cream" }
    - { amount: "8 oz", item: "paneer, cubed" }
    - { amount: "1/2 tbsp", item: "dried fenugreek leaves (kasuri methi), crushed" }
    - { amount: "To taste", item: "lemon juice" }
  steps:
    - "Blanch the spinach in salted boiling water for 2-3 minutes until wilted, then move it to ice water to keep the color."
    - "Drain and blend the spinach with the tomato, the 3 whole garlic cloves, ginger, green chili and cashews (if using) into a smooth puree."
    - "Heat the oil in a wide, heavy pan over medium heat and cook the chopped garlic until it changes color. Add the onion and cook 2-3 minutes until soft."
    - "Stir in the spinach puree and the water, cover and cook 10 minutes on medium heat, stirring so it doesn't stick."
    - "Add the garam masala, turmeric, chili powder and salt and cook 1 minute."
    - "Stir in the cream and paneer and simmer 3 minutes on low. Off the heat, stir in the fenugreek and lemon juice."
    - "Serve with naan, roti or rice."
note: "Adapted from Cook With Manali's palak paneer (4.94 stars from 224 votes). A pinch of baking soda in the pot helps keep the spinach bright green."
