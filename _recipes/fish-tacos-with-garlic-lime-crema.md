  prepTimeMinutes: 30

  servingsUnit: Tacos
  ingredients:
    - { amount: "24", item: "small white corn tortillas" }
    - { amount: "1 1/2 lbs", item: "tilapia" }
    - { amount: "1/2 tsp", item: "ground cumin" }
    - { amount: "1/2 tsp", item: "cayenne pepper" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "1/4 tsp", item: "black pepper" }
    - { amount: "1 tbsp", item: "olive oil" }
    - { amount: "1 tbsp", item: "unsalted butter" }
    - { amount: "1/2", item: "small purple cabbage, shredded" }
    - { amount: "2", item: "medium avocados, sliced" }
    - { amount: "2", item: "Roma tomatoes, diced (optional)" }
    - { amount: "1/2", item: "red onion, diced" }
    - { amount: "1/2 bunch", item: "cilantro, longer stems removed" }
    - { amount: "4 oz", item: "cotija cheese, grated" }
    - { amount: "1", item: "lime, cut into 8 wedges" }
    - { amount: "1/2 cup", item: "sour cream" }
    - { amount: "1/3 cup", item: "mayonnaise" }
    - { amount: "2 tbsp", item: "lime juice" }
    - { amount: "1 tsp", item: "garlic powder" }
    - { amount: "1 tsp", item: "Sriracha, or to taste" }
  steps:
    - "Heat the oven to 375°F and line a baking sheet with parchment. Mix the cumin, cayenne, salt and pepper and sprinkle over both sides of the fish."
    - "Drizzle the fish with olive oil, dot with butter and bake 20-25 minutes. Broil for 3-5 minutes at the end if you'd like browned edges."
    - "Whisk the sour cream, mayonnaise, lime juice, garlic powder and Sriracha for the sauce."
    - "Toast the tortillas quickly in a hot dry skillet."
    - "Flake the fish into the tortillas and top with cabbage, avocado, tomato, onion, cilantro, cotija and plenty of sauce. Serve with lime wedges."
note: "Adapted from Natasha's Kitchen's fish tacos (4.99 stars from 2,028 votes). The recipe is written for a crowd; halve it for a smaller group."
