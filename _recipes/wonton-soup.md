  prepTimeMinutes: 20
  cookTimeMinutes: 15
  servingsNumber: 6
  servingsUnit: Bowls
  ingredients:
    - { amount: "50-60", item: "wonton wrappers" }
    - { amount: "7 oz", item: "lean ground pork" }
    - { amount: "7 oz", item: "peeled shrimp, roughly chopped" }
    - { amount: "1 tbsp", item: "finely grated ginger" }
    - { amount: "2", item: "shallots or green onions, finely chopped" }
    - { amount: "1 tbsp", item: "light soy sauce" }
    - { amount: "2 tbsp", item: "Shaoxing wine" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "2 tbsp", item: "toasted sesame oil" }
    - { amount: "3 cups", item: "chicken broth" }
    - { amount: "2", item: "garlic cloves, smashed" }
    - { amount: "1/3 inch", item: "slice of ginger (optional)" }
    - { amount: "1 1/2 tbsp", item: "light soy sauce, for the broth" }
    - { amount: "2 tsp", item: "sugar" }
    - { amount: "1 1/2 tbsp", item: "Chinese cooking wine" }
    - { amount: "1/4-1/2 tsp", item: "sesame oil" }
    - { amount: "To serve", item: "chopped scallions, quartered bok choy and dried egg noodles (optional)" }
  steps:
    - "Mash the filling ingredients together with a potato masher about 20 times until fairly smooth, but leave small chunks of shrimp."
    - "Put 2 teaspoons of filling in the center of a wrapper, brush two edges with water, fold to seal, pressing out the air, then brush a corner with water and bring the two corners together, pressing to seal. Keep the wontons covered."
    - "Simmer the broth ingredients (broth, garlic, ginger, soy sauce, sugar, wine and sesame oil) in a saucepan 5-10 minutes, then pick out the garlic and ginger."
    - "Boil the wontons in a large pot of water 4 minutes until they float, and lift them straight into the bowls."
    - "Blanch the bok choy in the broth, and cook any noodles per the package."
    - "Ladle the hot broth over the wontons and finish with scallions."
note: "Adapted from RecipeTin Eats' wonton soup (4.97 stars from 261 votes). Uncooked wontons freeze well and cook from frozen in 6-8 minutes, unless you used thawed shrimp."
