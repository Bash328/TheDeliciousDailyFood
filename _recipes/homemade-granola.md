recipeContent:
  prepTimeMinutes: 10
  cookTimeMinutes: 30
  servingsNumber: 12
  servingsUnit: Servings
  ingredients:
    - { amount: "4 cups", item: "old-fashioned rolled oats" }
    - { amount: "1 cup", item: "chopped nuts (almonds, pecans)" }
    - { amount: "1/2 cup", item: "honey or maple syrup" }
    - { amount: "1/3 cup", item: "coconut oil or vegetable oil" }
    - { amount: "1 tsp", item: "ground cinnamon" }
    - { amount: "1/2 tsp", item: "salt" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "1 cup", item: "dried fruit (raisins, cranberries)" }
  steps:
    - "Heat the oven to 325°F and line a baking sheet with parchment."
    - "Mix the oats, nuts, cinnamon and salt. Stir in the honey, oil and vanilla until evenly coated."
    - "Spread in a single layer and press it down firmly."
    - "Bake 25-30 minutes, stirring once halfway, until golden."
    - "Cool completely without stirring, so it forms clusters, then add the dried fruit and break apart."
note: "Store in an airtight container for up to 2 weeks."
