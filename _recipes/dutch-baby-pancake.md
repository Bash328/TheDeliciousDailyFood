  ingredients:
    - { amount: "3", item: "large eggs" }
    - { amount: "3/4 cup", item: "milk" }
    - { amount: "3/4 cup", item: "all-purpose flour" }
    - { amount: "1 tbsp", item: "granulated sugar" }
    - { amount: "1/4 tsp", item: "salt" }
    - { amount: "1 tsp", item: "vanilla extract" }
    - { amount: "3 tbsp", item: "butter" }
    - { amount: "To serve", item: "powdered sugar, lemon wedges and berries" }
  steps:
    - "Heat the oven to 425°F with a 10-inch oven-safe skillet inside."
    - "Blend the eggs, milk, flour, sugar, salt and vanilla until smooth."
    - "Add the butter to the hot skillet, swirl to coat, and pour in the batter."
    - "Bake 18-20 minutes until puffed and deep golden."
    - "Dust with powdered sugar and serve right away with lemon and berries, before it deflates."
