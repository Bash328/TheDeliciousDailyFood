image: ""
recipeContent:
  prepTimeMinutes: 25
  cookTimeMinutes: 40
  servingsNumber: 16
  servingsUnit: Slices
  ingredients:
    - { amount: "1 1/4 cups", item: "vegetable oil" }
    - { amount: "2 cups", item: "granulated sugar" }
    - { amount: "1 tsp", item: "salt" }
    - { amount: "4", item: "large eggs" }
    - { amount: "1 tbsp", item: "ground cinnamon" }
    - { amount: "1/2 tsp", item: "ground ginger" }
    - { amount: "1/4 tsp", item: "ground cloves" }
    - { amount: "1 1/2 tsp", item: "baking powder" }
    - { amount: "1/2 tsp", item: "baking soda" }
    - { amount: "2 cups", item: "all-purpose flour" }
    - { amount: "3 cups", item: "finely grated carrots (about 326g)" }
    - { amount: "1 1/2 cups", item: "chopped pecans or walnuts, toasted" }
    - { amount: "1/2 cup (8 tbsp)", item: "unsalted butter, room temperature, for the frosting" }
    - { amount: "8 oz", item: "cream cheese, room temperature" }
    - { amount: "1/4 tsp", item: "salt, for the frosting" }
    - { amount: "2 tsp", item: "vanilla extract" }
    - { amount: "2 1/2 to 3 1/2 cups", item: "powdered sugar" }
  steps:
    - "Heat the oven to 350°F and grease two 9-inch round pans (or one 9x13-inch pan). Toast the nuts on a baking sheet for 6-8 minutes if you haven't already."
    - "Beat the oil, sugar, salt, eggs, cinnamon, ginger, cloves, baking powder and baking soda until smooth."
    - "Stir in the flour until just blended, then fold in the carrots and nuts."
    - "Divide the batter between the pans and bake 35-40 minutes, until a toothpick comes out clean or with a few moist crumbs."
    - "Cool the layers 15 minutes in the pans, then turn out onto a rack and cool completely before frosting."
    - "For the frosting, beat the butter with the cream cheese until smooth, then add the salt and vanilla. Beat in the powdered sugar a little at a time until spreadable, adding a teaspoon of milk if too stiff."
    - "Frost between the layers and over the top and sides. Chill, then let sit at room temperature for an hour before serving."
note: "Proportions adapted from King Arthur Baking's carrot cake (4.6 stars, 431 reviews). Keeps about a day in the fridge, well wrapped, or freeze for longer."
