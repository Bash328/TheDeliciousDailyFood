category: Dinner
cuisine: "American"
description: "Fall-apart beef with carrots and potatoes in a savory gravy."
image: ""
recipeContent:
  prepTimeMinutes: 20
  cookTimeMinutes: 210
  servingsNumber: 6
  servingsUnit: Servings
  ingredients:
    - { amount: "3 lb", item: "beef chuck roast" }
    - { amount: "1 1/2 tsp", item: "salt" }
    - { amount: "1 tsp", item: "black pepper" }
    - { amount: "2 tbsp", item: "olive oil" }
    - { amount: "1", item: "large onion, sliced" }
    - { amount: "4", item: "garlic cloves, smashed" }
    - { amount: "2 tbsp", item: "tomato paste" }
    - { amount: "2 cups", item: "beef broth" }
    - { amount: "1 tbsp", item: "Worcestershire sauce" }
    - { amount: "2", item: "sprigs fresh thyme" }
    - { amount: "4", item: "carrots, cut into chunks" }
    - { amount: "1 1/2 lbs", item: "baby potatoes" }
  steps:
    - "Heat the oven to 325°F. Season the roast all over with salt and pepper."
    - "Sear in the oil in a Dutch oven over medium-high heat, 4 minutes per side. Remove."
    - "Cook the onion 4 minutes, add the garlic and tomato paste for 1 minute, then add the broth, Worcestershire and thyme."
    - "Return the roast to the pot, cover and roast 2 1/2 hours."
    - "Add the carrots and potatoes around the roast, cover and cook 1 hour more until the meat shreds easily."
    - "Slice or shred the beef and serve with the vegetables and the juices spooned over."
note: "If the juices are thin, simmer them on the stove for 5 minutes, or whisk in a spoonful of flour mixed with water."
