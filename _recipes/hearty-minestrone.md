    - { amount: "2 tbsp", item: "olive oil" }
    - { amount: "1", item: "onion, diced" }
    - { amount: "2", item: "carrots, diced" }
    - { amount: "2", item: "celery stalks, diced" }
    - { amount: "3", item: "garlic cloves, minced" }
    - { amount: "1 can (14 oz)", item: "diced tomatoes" }
    - { amount: "6 cups", item: "vegetable broth" }
    - { amount: "1 can (15 oz)", item: "cannellini beans, drained" }
    - { amount: "1", item: "zucchini, diced" }
    - { amount: "1 tsp", item: "dried Italian seasoning" }
    - { amount: "1 cup", item: "small pasta (ditalini or elbows)" }
    - { amount: "2 cups", item: "fresh spinach" }
    - { amount: "1/2 cup", item: "grated Parmesan" }
  steps:
    - "Cook the onion, carrots and celery in the oil for 6 minutes. Add the garlic for 30 seconds."
    - "Add the tomatoes, broth, beans, zucchini and seasoning and simmer 20 minutes."
    - "Add the pasta and cook until tender, 8-10 minutes. Stir in the spinach until wilted."
    - "Serve topped with Parmesan."
