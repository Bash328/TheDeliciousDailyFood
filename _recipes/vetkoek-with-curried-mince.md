  prepTimeMinutes: 30

  servingsNumber: 8

  ingredients:
    - { amount: "2 tsp", item: "salt" }
    - { amount: "2-3 tbsp", item: "sugar" }
    - { amount: "2 1/4 tsp", item: "instant yeast (1 packet)" }
    - { amount: "2 cups", item: "lukewarm water" }
    - { amount: "5 1/2 cups", item: "cake flour or all-purpose flour, plus more as needed" }
    - { amount: "For frying", item: "vegetable oil, about 3 inches deep" }
    - { amount: "2 tbsp", item: "canola oil, for the mince" }
    - { amount: "1", item: "onion, sliced" }
    - { amount: "1 1/2 tsp", item: "garlic, minced" }
    - { amount: "1 tsp", item: "ground ginger" }
    - { amount: "2 tsp", item: "curry powder" }
    - { amount: "1/2 tsp", item: "paprika" }
    - { amount: "2-3", item: "tomatoes, chopped" }
    - { amount: "2 cups", item: "potatoes, diced" }
    - { amount: "1 lb", item: "ground beef" }
    - { amount: "1/2 tsp", item: "white pepper" }
    - { amount: "1 cup", item: "frozen peas and carrots" }
    - { amount: "2-3 tbsp", item: "parsley or cilantro, chopped" }
    - { amount: "To taste", item: "salt" }
  steps:
    - "Mix the salt, sugar, yeast and water and let stand 5 minutes. Stir in the flour until a dough forms, cover loosely and let rise 1-2 hours until doubled."
    - "Meanwhile make the mince. Cook the onion, garlic, ginger, curry powder and paprika in the oil for 2-3 minutes, stirring constantly."
    - "Add the tomatoes and cook 3 minutes, then add the potatoes, beef and white pepper. Simmer about 15 minutes until the potatoes are tender."
    - "Stir in the peas and carrots for a minute or two, then the parsley, and season with salt. This keeps a day ahead and tastes even better."
    - "Knock the air out of the dough on a floured surface and knead until firm. Roll about 1 inch thick and cut out rounds with a 3-inch cutter."
    - "Heat the frying oil to 350°F and fry the vetkoek 3-5 minutes, until deep golden. Drain on paper towels."
    - "Split each vetkoek and fill it with the warm mince."
note: "Adapted from African Bites' vetkoek (4.98 stars from 35 votes)."
