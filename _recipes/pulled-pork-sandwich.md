  ingredients:
    - { amount: "4 lb", item: "boneless pork shoulder" }
    - { amount: "2 tbsp", item: "brown sugar" }
    - { amount: "1 tbsp", item: "smoked paprika" }
    - { amount: "1 tbsp", item: "garlic powder" }
    - { amount: "1 tbsp", item: "salt" }
    - { amount: "1 tsp", item: "black pepper" }
    - { amount: "1 cup", item: "barbecue sauce" }
    - { amount: "1/2 cup", item: "apple cider vinegar" }
    - { amount: "10", item: "hamburger buns" }
    - { amount: "To serve", item: "coleslaw" }
  steps:
    - "Mix the sugar, paprika, garlic powder, salt and pepper and rub all over the pork."
    - "Place in a slow cooker with the vinegar. Cook on low 8-10 hours until the meat shreds easily."
    - "Shred the pork with two forks and stir in the barbecue sauce and 1/2 cup of the cooking juices."
    - "Pile onto toasted buns and top with coleslaw."
note: "For the oven: cook covered in a Dutch oven at 300°F for 4 to 5 hours."
